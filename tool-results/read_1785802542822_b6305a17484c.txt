   135→// Map any unknown string status into one of our 5 known buckets so the

   137→function coerceStatus(raw?: string | null): VideoProjectStatus {
   138→  if (!raw) return 'producing'
   139→  if (raw in STATUS_CONFIG) return raw as VideoProjectStatus
   140→  // Reasonable fallbacks for legacy / pipeline-internal statuses.

   143→  }
   144→  if (raw === 'uploading') return 'uploaded'
   145→  return 'producing'
   146→}

   185→function toTimestamp(value?: string | Date | null): number {
   186→  if (!value) return 0
   187→  const d = value instanceof Date ? value : new Date(value)
   188→  const t = d.getTime()
   189→  return Number.isNaN(t) ? 0 : t
   190→}
   191→
   192→function relativeTime(value?: string | Date | null): string {
   193→  const t = toTimestamp(value)
   194→  if (!t) return '—'
   195→  const diff = Date.now() - t
   196→  if (diff < 0) return 'just now'
   197→  const sec = Math.floor(diff / 1000)
   198→  if (sec < 60) return `${sec}s ago`
   199→  const min = Math.floor(sec / 60)
   200→  if (min < 60) return `${min}m ago`
   201→  const hr = Math.floor(min / 60)
   202→  if (hr < 24) return `${hr}h ago`
   203→  const day = Math.floor(hr / 24)
   204→  if (day < 7) return `${day}d ago`
   205→  const week = Math.floor(day / 7)
   206→  if (week < 4) return `${week}w ago`
   207→  const month = Math.floor(day / 30)
   208→  if (month < 12) return `${month}mo ago`
   209→  const year = Math.floor(day / 365)
   210→  return `${year}y ago`
   211→}
   212→
   213→function formatDuration(seconds?: number | null): string {
   214→  if (seconds === null || seconds === undefined || Number.isNaN(seconds)) {
   215→    return '—'
   216→  }
   217→  if (seconds < 60) return `${seconds.toFixed(0)}s`
   218→  const m = Math.floor(seconds / 60)
   219→  const s = Math.floor(seconds % 60)
   220→  return `${m}:${s.toString().padStart(2, '0')}`
   221→}
   222→
   223→function formatFileSize(bytes?: number | null): string {
   224→  if (!bytes || bytes <= 0) return '—'
   225→  if (bytes < 1024) return `${bytes} B`
   226→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   227→  if (bytes < 1024 * 1024 * 1024) return `${(bytes / 1024 / 1024).toFixed(1)} MB`
   228→  return `${(bytes / 1024 / 1024 / 1024).toFixed(2)} GB`
   229→}
   230→
   231→function truncate(text: string, max: number): string {
   232→  if (text.length <= max) return text
   233→  return text.slice(0, max - 1).trimEnd() + '…'
   234→}

   238→export function VideoProjectExplorer({
   239→  projects,
   240→  onPreview,
   241→  onBulkAction,
   242→  className,
   243→}: VideoProjectExplorerProps) {
   244→  const { toast } = useToast()
   245→  const [searchInput, setSearchInput] = useState('')
   246→  const [debouncedSearch, setDebouncedSearch] = useState('')
   247→  const [statusFilter, setStatusFilter] = useState<'all' | VideoProjectStatus>(
   248→    'all',
   249→  )
   250→  const [sortKey, setSortKey] = useState<SortKey>('updated_desc')
   251→
   252→  // ─── Bulk selection state ──────────────────────────────────────────
   253→  const [selectMode, setSelectMode] = useState(false)
   254→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
   255→  const [bulkLoading, setBulkLoading] = useState<BulkActionType | null>(null)
   256→  const [confirmDialog, setConfirmDialog] = useState<
   257→    null | 'delete' | 're-render'
   258→  >(null)
   259→
   260→  // Debounced search (200ms)
   261→  useEffect(() => {
   262→    const handle = setTimeout(() => {
   263→      setDebouncedSearch(searchInput.trim().toLowerCase())
   264→    }, 200)
   265→    return () => clearTimeout(handle)
   266→  }, [searchInput])
   267→
   268→  const filteredProjects = useMemo(() => {
   269→    const out: VideoProject[] = []
   270→    for (const project of projects) {
   271→      const title = (project.title ?? '').toLowerCase()
   272→      if (debouncedSearch && !title.includes(debouncedSearch)) continue
   273→      if (statusFilter !== 'all') {
   274→        const s = coerceStatus(project.status)
   275→        if (s !== statusFilter) continue
   276→      }
   277→      out.push(project)
   278→    }
   279→
   280→    out.sort((a, b) => {
   281→      switch (sortKey) {
   282→        case 'updated_asc':
   283→          return toTimestamp(a.updatedAt) - toTimestamp(b.updatedAt)
   284→        case 'updated_desc':
   285→          return toTimestamp(b.updatedAt) - toTimestamp(a.updatedAt)
   286→        case 'duration_asc': {
   287→          const da = a.duration ?? 0
   288→          const db = b.duration ?? 0
   289→          return da - db
   290→        }
   291→        case 'duration_desc': {
   292→          const da = a.duration ?? 0
   293→          const db = b.duration ?? 0
   294→          return db - da
   295→        }
   296→        case 'status': {
   297→          const sa = coerceStatus(a.status)
   298→          const sb = coerceStatus(b.status)
   299→          return sa.localeCompare(sb)
   300→        }
   301→        default:
   302→          return 0
   303→      }
   304→    })
   305→    return out
   306→  }, [projects, debouncedSearch, statusFilter, sortKey])
   307→
   308→  const activeFilterCount = useMemo(() => {
   309→    let n = 0
   310→    if (debouncedSearch) n++
   311→    if (statusFilter !== 'all') n++
   312→    return n
   313→  }, [debouncedSearch, statusFilter])
   314→
   315→  // Prune stale IDs from selection when the projects list changes
   316→  // (e.g. after a bulk delete or after the parent refetches).
   317→  useEffect(() => {
   318→    if (selectedIds.size === 0) return
   319→    const validIds = new Set(projects.map((p) => p.id))
   320→    let changed = false
   321→    const next = new Set<string>()
   322→    for (const id of selectedIds) {
   323→      if (validIds.has(id)) {
   324→        next.add(id)
   325→      } else {
   326→        changed = true
   327→      }
   328→    }
   329→    if (changed) {
   330→      setSelectedIds(next)
   331→      if (next.size === 0) setSelectMode(false)
   332→    }
   333→  }, [projects, selectedIds])
   334→
   335→  const clearFilters = useCallback(() => {
   336→    setSearchInput('')
   337→    setDebouncedSearch('')
   338→    setStatusFilter('all')
   339→    setSortKey('updated_desc')
   340→  }, [])
   341→
   342→  // ─── Bulk action handlers ──────────────────────────────────────────
   343→  const toggleSelectMode = useCallback(() => {
   344→    setSelectMode((prev) => {
   345→      const next = !prev
   346→      if (!next) setSelectedIds(new Set())
   347→      return next
   348→    })
   349→  }, [])
   350→
   351→  const toggleSelect = useCallback((id: string) => {
   352→    setSelectedIds((prev) => {
   353→      const next = new Set(prev)
   354→      if (next.has(id)) next.delete(id)
   355→      else next.add(id)
   356→      return next
   357→    })
   358→  }, [])
   359→
   360→  const selectAllFiltered = useCallback(() => {
   361→    setSelectedIds(new Set(filteredProjects.map((p) => p.id)))
   362→  }, [filteredProjects])
   363→
   364→  const clearSelection = useCallback(() => {
   365→    setSelectedIds(new Set())
   366→  }, [])
   367→
   368→  const executeBulkAction = useCallback(
   369→    async (
   370→      action: BulkActionType,
   371→      payload?: Record<string, unknown>,
   372→    ) => {
   373→      if (selectedIds.size === 0) return
   374→      setBulkLoading(action)
   375→      const ids = Array.from(selectedIds)
   376→      try {
   377→        const res = await fetch('/api/data/projects/bulk', {
   378→          method: 'POST',
   379→          headers: { 'Content-Type': 'application/json' },
   380→          body: JSON.stringify({ action, projectIds: ids, payload }),
   381→        })
   382→        const data = (await res.json().catch(() => ({}))) as {
   383→          ok?: boolean
   384→          affected?: number
   385→          error?: string
   386→        }
   387→        if (!res.ok || !data.ok) {
   388→          throw new Error(
   389→            data.error || `Bulk ${action} failed (HTTP ${res.status})`,
   390→          )
   391→        }
   392→        const affected = data.affected ?? 0
   393→        toast({
   394→          type: 'success',
   395→          title: `Bulk ${action} complete`,
   396→          description: `${affected} project${affected === 1 ? '' : 's'} affected.`,
   397→        })
   398→        // Clear local selection first, then notify parent so it can refresh.
   399→        setSelectedIds(new Set())
   400→        setSelectMode(false)
   401→        setConfirmDialog(null)
   402→        onBulkAction?.()
   403→      } catch (err) {
   404→        toast({
   405→          type: 'error',
   406→          title: `Bulk ${action} failed`,
   407→          description: err instanceof Error ? err.message : 'Unknown error',
   408→          duration: 5000,
   409→        })
   410→      } finally {
   411→        setBulkLoading(null)
   412→      }
   413→    },
   414→    [selectedIds, toast, onBulkAction],
   415→  )
   416→
   417→  const handleCardClick = useCallback(
   418→    (id: string) => {
   419→      if (selectMode) {
   420→        toggleSelect(id)
   421→        return
   422→      }
   423→      onPreview?.(id)
   424→    },
   425→    [selectMode, toggleSelect, onPreview],
   426→  )
   427→
   428→  const isLoading = projects.length === 0
   429→
   430→  // ─── Render ────────────────────────────────────────────────────────
   431→
   432→  return (
   433→    <div
   434→      className={cn(
   435→        'w-full rounded-xl bg-slate-950 p-3 sm:p-4',
   436→        'border border-slate-800/50',
   437→        className,
   438→      )}
   439→    >
   440→      {/* ─── Header / Search ─── */}
   441→      <div className="mb-3 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   442→        <div className="flex items-center gap-2">
   443→          <div className="flex size-8 items-center justify-center rounded-lg bg-gradient-to-br from-emerald-500/20 to-cyan-500/20 border border-emerald-500/30">
   444→            <Film className="size-4 text-emerald-300" />
   445→          </div>
   446→          <div>
   447→            <h3 className="text-sm font-semibold text-slate-100">
   448→              Video Project Explorer
   449→            </h3>
   450→            <p className="text-[11px] text-slate-400">
   451→              Produce, approve &amp; manage rendered videos
   452→            </p>
   453→          </div>
   454→        </div>
   455→
   456→        <div className="flex items-center gap-2">
   457→          <div className="relative w-full sm:w-72">
   458→            <Search className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-slate-500" />
   459→            <Input
   460→              value={searchInput}
   461→              onChange={(e) => setSearchInput(e.target.value)}
   462→              placeholder="Search projects by title…"
   463→              className="h-9 border-slate-700/60 bg-slate-900/60 pl-9 pr-8 text-sm text-slate-100 placeholder:text-slate-500 focus-visible:border-emerald-500/50 focus-visible:ring-emerald-500/20"
   464→            />
   465→            {searchInput && (
   466→              <button
   467→                type="button"
   468→                onClick={() => setSearchInput('')}
   469→                className="absolute right-2 top-1/2 -translate-y-1/2 rounded p-1 text-slate-500 hover:bg-slate-800 hover:text-slate-300"
   470→                aria-label="Clear search"
   471→              >
   472→                <X className="size-3.5" />
   473→              </button>
   474→            )}
   475→          </div>
   476→
   477→          {/* Select toggle */}
   478→          <Tooltip>
   479→            <TooltipTrigger asChild>
   480→              <Button
   481→                type="button"
   482→                variant={selectMode ? 'default' : 'outline'}
   483→                size="sm"
   484→                onClick={toggleSelectMode}
   485→                className={cn(
   486→                  'h-9 shrink-0 gap-1.5 px-3 text-xs',

   497→                <span className="hidden sm:inline">Select</span>

   499→            </TooltipTrigger>
   500→            <TooltipContent className="bg-slate-800 text-slate-100">

   503→          </Tooltip>
   504→        </div>

   507→      {/* ─── Filter Row: status pills + sort ─── */}
   508→      <div className="mb-3 flex flex-wrap items-center gap-2">

   510→        <div className="flex items-center rounded-lg border border-slate-700/60 bg-slate-900/40 p-0.5">

   512→            const isActive = statusFilter === p.value
   513→            return (
   514→              <button
   515→                key={p.value}
   516→                type="button"
   517→                onClick={() => setStatusFilter(p.value)}
   518→                className={cn(
   519→                  'relative rounded-md px-2.5 py-1 text-[11px] font-medium transition-colors',
   520→                  isActive
   521→                    ? 'text-white'
   522→                    : 'text-slate-400 hover:text-slate-200',
   523→                )}
   524→              >
   525→                {isActive && (
   526→                  <motion.span
   527→                    layoutId="vpx-status-pill-bg"
   528→                    className="absolute inset-0 rounded-md bg-gradient-to-r from-emerald-500/80 to-cyan-500/80"
   529→                    transition={{ type: 'spring', duration: 0.3, bounce: 0.2 }}
   530→                  />
   531→                )}
   532→                <span className="relative z-10">{p.label}</span>
   533→              </button>

   536→        </div>

   538→        {/* Sort */}

   556→          </SelectContent>
   557→        </Select>
   558→
   559→        {/* Active filter count */}
   560→        {activeFilterCount > 0 && (
