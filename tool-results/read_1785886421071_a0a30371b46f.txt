   231→// ─── Animated Counter Hook ───────────────────────────────────────────
   232→function useAnimatedCounter(end: number, duration: number = 800) {
   233→  const [count, setCount] = useState(end)
   234→  const prevEnd = useRef(end)
   235→  useEffect(() => {
   236→    if (prevEnd.current === end) return
   237→    const start = prevEnd.current
   238→    const startTime = performance.now()
   239→    const step = (now: number) => {
   240→      const progress = Math.min((now - startTime) / duration, 1)
   241→      const eased = 1 - Math.pow(1 - progress, 3) // easeOutCubic
   242→      setCount(Math.round(start + (end - start) * eased))
   243→      if (progress < 1) requestAnimationFrame(step)
   244→    }
   245→    requestAnimationFrame(step)
   246→    prevEnd.current = end
   247→    return () => {}
   248→  }, [end, duration])
   249→  return count
   250→}

   313→function PipelineFlow({ pipeline }: { pipeline: AgentStatus['pipeline'] | null }) {
   314→  if (!pipeline) return null
   315→  const total = pipeline ? Object.values(pipeline).reduce((a: number, b: number) => a + b, 0) : 0
   316→  return (
   317→    <div className="flex items-center gap-0 overflow-x-auto py-2 px-1">
   318→      {PIPELINE_STAGES.map((stage, i) => {
   319→        const count = pipeline[stage.key as keyof typeof pipeline] || 0
   320→        const pct = total > 0 ? Math.round((count / total) * 100) : 0
   321→        const Icon = stage.icon
   322→        const isEmpty = count === 0
   323→        return (
   324→          <React.Fragment key={stage.key}>
   325→            <motion.div
   326→              initial={{ opacity: 0, scale: 0.8 }}
   327→              animate={{ opacity: 1, scale: 1 }}
   328→              transition={{ delay: i * 0.08, duration: 0.3 }}
   329→              className={`flex flex-col items-center min-w-[80px] p-3 rounded-xl border ${stage.border} ${stage.bg} transition-all duration-300 hover:scale-105 hover:shadow-lg ${isEmpty ? 'opacity-60' : ''}`}
   330→            >
   331→              <Icon className={`w-5 h-5 ${stage.textColor} mb-1`} />
   332→              <span className={`text-xl font-bold ${stage.textColor} font-tabular-nums`}>{isEmpty ? '+' : count}</span>
   333→              <span className="text-[10px] text-slate-400 mt-0.5">{stage.label}</span>
   334→              {pct > 0 && (
   335→                <span className="text-[10px] text-slate-400 font-mono mt-0.5">{pct}%</span>
   336→              )}
   337→            </motion.div>
   338→            {i < PIPELINE_STAGES.length - 1 && (
   339→              <motion.div
   340→                initial={{ opacity: 0, x: -8 }}
   341→                animate={{ opacity: 1, x: 0 }}
   342→                transition={{ delay: i * 0.08 + 0.04, duration: 0.3 }}
   343→                className="flex items-center px-1"
   344→              >
   345→                <ChevronRight className="w-4 h-4 text-slate-500 group-hover/stage:text-slate-400 transition-colors" />
   346→              </motion.div>
   347→            )}
   348→          </React.Fragment>
   349→        )
   350→      })}
   351→    </div>
   352→  )
   353→}
