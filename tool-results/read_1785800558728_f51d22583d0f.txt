   176→  // New categorized actions
   177→  if (action === 'emergency_stop') return 'text-red-400 bg-red-500/10 border-red-500/30'
   178→  if (action === 'mode_change') return 'text-amber-400 bg-amber-500/10 border-amber-500/30'
   179→  if (action === 'upload') return 'text-cyan-400 bg-cyan-500/10 border-cyan-500/30'
   180→  if (action === 'strategy_change') return 'text-violet-400 bg-violet-500/10 border-violet-500/30'
   181→  // Legacy / fallback keyword matching
   182→  if (action.includes('error') || action.includes('fail')) return 'text-red-400 bg-red-500/10 border-red-500/30'
   183→  if (action.includes('upload') || action.includes('complete')) return 'text-emerald-400 bg-emerald-500/10 border-emerald-500/30'

   218→// ─── Animated Counter Hook ───────────────────────────────────────────
   219→function useAnimatedCounter(end: number, duration: number = 800) {
   220→  const [count, setCount] = useState(end)
   221→  const prevEnd = useRef(end)
   222→  useEffect(() => {
   223→    if (prevEnd.current === end) return
   224→    const start = prevEnd.current
   225→    const startTime = performance.now()
   226→    const step = (now: number) => {
   227→      const progress = Math.min((now - startTime) / duration, 1)
   228→      const eased = 1 - Math.pow(1 - progress, 3) // easeOutCubic
   229→      setCount(Math.round(start + (end - start) * eased))
   230→      if (progress < 1) requestAnimationFrame(step)
   231→    }
   232→    requestAnimationFrame(step)
   233→    prevEnd.current = end
   234→    return () => {}
   235→  }, [end, duration])
   236→  return count
   237→}

   258→function StatusCard({ icon: Icon, label, value, sub, trend, color = 'text-emerald-400', valueSuffix, hint }: {
   259→  icon: any; label: string; value: string | number; sub?: string; trend?: 'up' | 'down' | 'flat'; color?: string; valueSuffix?: string; hint?: string
   260→}) {
   261→  const numericValue = typeof value === 'number' ? value : 0
   262→  const animatedValue = useAnimatedCounter(numericValue)
   263→  const displayValue = typeof value === 'number' ? animatedValue : value
   264→  const bgForColor = (c: string) => c === 'text-emerald-400' ? 'bg-emerald-500/10' : c === 'text-blue-400' ? 'bg-blue-500/10' : c === 'text-amber-400' ? 'bg-amber-500/10' : c === 'text-violet-400' ? 'bg-violet-500/10' : c === 'text-cyan-400' ? 'bg-cyan-500/10' : c === 'text-rose-400' ? 'bg-rose-500/10' : 'bg-slate-500/10'
   265→  const trendConfig = trend === 'up' ? { icon: ArrowUpRight, color: 'text-emerald-400', bg: 'bg-emerald-500/20' } : trend === 'down' ? { icon: ArrowDownRight, color: 'text-red-400', bg: 'bg-red-500/20' } : trend === 'flat' ? { icon: Minus, color: 'text-slate-400', bg: 'bg-slate-500/20' } : null
   266→  return (
   267→    <GradientCard>
   268→      <div className="p-4 group cursor-default" title={hint}>
   269→        <div className="flex items-start justify-between">
   270→          <div className={`p-2.5 rounded-xl ${bgForColor(color)} transition-all duration-300 group-hover:scale-110 group-hover:shadow-lg`}>
   271→            <Icon className={`w-5 h-5 ${color}`} />
   272→          </div>
   273→          {trendConfig && (

   292→function PipelineFlow({ pipeline }: { pipeline: AgentStatus['pipeline'] | null }) {
   293→  if (!pipeline) return null
   294→  const total = pipeline ? Object.values(pipeline).reduce((a: number, b: number) => a + b, 0) : 0
   295→  return (
   296→    <div className="flex items-center gap-0 overflow-x-auto py-2 px-1">
   297→      {PIPELINE_STAGES.map((stage, i) => {
   298→        const count = pipeline[stage.key as keyof typeof pipeline] || 0
   299→        const pct = total > 0 ? Math.round((count / total) * 100) : 0
   300→        const Icon = stage.icon
   301→        return (
   302→          <React.Fragment key={stage.key}>
   303→            <motion.div
   304→              initial={{ opacity: 0, scale: 0.8 }}
   305→              animate={{ opacity: 1, scale: 1 }}
   306→              transition={{ delay: i * 0.08, duration: 0.3 }}
   307→              className={`flex flex-col items-center min-w-[80px] p-3 rounded-xl border ${stage.border} ${stage.bg} transition-all duration-300 hover:scale-105 hover:shadow-lg`}
   308→            >
   309→              <Icon className={`w-5 h-5 ${stage.textColor} mb-1`} />
   310→              <span className={`text-xl font-bold ${stage.textColor}`}>{count}</span>
   311→              <span className="text-[10px] text-slate-400 mt-0.5">{stage.label}</span>
   312→              {pct > 0 && (
   313→                <span className="text-[9px] text-slate-500 font-mono mt-0.5">{pct}%</span>
   314→              )}
   315→            </motion.div>
   316→            {i < PIPELINE_STAGES.length - 1 && (
   317→              <motion.div
   318→                initial={{ opacity: 0, x: -8 }}
   319→                animate={{ opacity: 1, x: 0 }}
   320→                transition={{ delay: i * 0.08 + 0.04, duration: 0.3 }}
   321→                className="flex items-center px-1"
   322→              >
   323→                <ChevronRight className="w-4 h-4 text-slate-600" />
   324→              </motion.div>
   325→            )}
   326→          </React.Fragment>
   327→        )
   328→      })}
   329→    </div>
   330→  )
   331→}
