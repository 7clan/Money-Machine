   178→  // New categorized actions
   179→  if (action === 'emergency_stop') return 'text-red-400 bg-red-500/10 border-red-500/30'
   180→  if (action === 'mode_change') return 'text-amber-400 bg-amber-500/10 border-amber-500/30'
   181→  if (action === 'upload') return 'text-cyan-400 bg-cyan-500/10 border-cyan-500/30'
   182→  if (action === 'strategy_change') return 'text-violet-400 bg-violet-500/10 border-violet-500/30'
   183→  // Legacy / fallback keyword matching
   184→  if (action.includes('error') || action.includes('fail')) return 'text-red-400 bg-red-500/10 border-red-500/30'
   185→  if (action.includes('upload') || action.includes('complete')) return 'text-emerald-400 bg-emerald-500/10 border-emerald-500/30'
   186→  if (action.includes('produce') || action.includes('render')) return 'text-cyan-400 bg-cyan-500/10 border-cyan-500/30'

   220→// ─── Animated Counter Hook ───────────────────────────────────────────
   221→function useAnimatedCounter(end: number, duration: number = 800) {
   222→  const [count, setCount] = useState(end)
   223→  const prevEnd = useRef(end)
   224→  useEffect(() => {
   225→    if (prevEnd.current === end) return
   226→    const start = prevEnd.current
   227→    const startTime = performance.now()
   228→    const step = (now: number) => {
   229→      const progress = Math.min((now - startTime) / duration, 1)
   230→      const eased = 1 - Math.pow(1 - progress, 3) // easeOutCubic
   231→      setCount(Math.round(start + (end - start) * eased))
   232→      if (progress < 1) requestAnimationFrame(step)
   233→    }
   234→    requestAnimationFrame(step)
   235→    prevEnd.current = end
   236→    return () => {}
   237→  }, [end, duration])
   238→  return count
   239→}

   260→function StatusCard({ icon: Icon, label, value, sub, trend, color = 'text-emerald-400', valueSuffix, hint }: {
   261→  icon: any; label: string; value: string | number; sub?: string; trend?: 'up' | 'down' | 'flat'; color?: string; valueSuffix?: string; hint?: string
   262→}) {
   263→  const numericValue = typeof value === 'number' ? value : 0
   264→  const animatedValue = useAnimatedCounter(numericValue)
   265→  const displayValue = typeof value === 'number' ? animatedValue : value
   266→  const bgForColor = (c: string) => c === 'text-emerald-400' ? 'bg-emerald-500/10' : c === 'text-blue-400' ? 'bg-blue-500/10' : c === 'text-amber-400' ? 'bg-amber-500/10' : c === 'text-violet-400' ? 'bg-violet-500/10' : c === 'text-cyan-400' ? 'bg-cyan-500/10' : c === 'text-rose-400' ? 'bg-rose-500/10' : 'bg-slate-500/10'
   267→  const trendConfig = trend === 'up' ? { icon: ArrowUpRight, color: 'text-emerald-400', bg: 'bg-emerald-500/20' } : trend === 'down' ? { icon: ArrowDownRight, color: 'text-red-400', bg: 'bg-red-500/20' } : trend === 'flat' ? { icon: Minus, color: 'text-slate-400', bg: 'bg-slate-500/20' } : null
   268→  return (
   269→    <GradientCard>
   270→      <div className="p-4 group cursor-default" title={hint}>
   271→        <div className="flex items-start justify-between">
   272→          <div className={`p-2.5 rounded-xl ${bgForColor(color)} transition-all duration-300 group-hover:scale-110 group-hover:shadow-lg`}>
   273→            <Icon className={`w-5 h-5 ${color}`} />
   274→          </div>
   275→          {trendConfig && (
   276→            <div className={`flex items-center gap-0.5 text-xs px-1.5 py-0.5 rounded-full ${trendConfig.bg} ${trendConfig.color}`}>
   277→              <trendConfig.icon className="w-3 h-3" />

   281→        <div className="mt-3">
   282→          <div className="flex items-baseline gap-1">

   284→            {valueSuffix && <span className={`text-sm font-semibold ${color}`}>{valueSuffix}</span>}
   285→          </div>
   286→          <p className="text-xs text-slate-400 mt-1 font-medium">{label}</p>
   287→          {sub && <p className="text-[10px] text-slate-500 mt-0.5 truncate" title={sub}>{sub}</p>}

   294→function PipelineFlow({ pipeline }: { pipeline: AgentStatus['pipeline'] | null }) {
   295→  if (!pipeline) return null
   296→  const total = pipeline ? Object.values(pipeline).reduce((a: number, b: number) => a + b, 0) : 0
   297→  return (
   298→    <div className="flex items-center gap-0 overflow-x-auto py-2 px-1">
   299→      {PIPELINE_STAGES.map((stage, i) => {
   300→        const count = pipeline[stage.key as keyof typeof pipeline] || 0
   301→        const pct = total > 0 ? Math.round((count / total) * 100) : 0
   302→        const Icon = stage.icon
   303→        return (
   304→          <React.Fragment key={stage.key}>
   305→            <motion.div
   306→              initial={{ opacity: 0, scale: 0.8 }}
   307→              animate={{ opacity: 1, scale: 1 }}
   308→              transition={{ delay: i * 0.08, duration: 0.3 }}
   309→              className={`flex flex-col items-center min-w-[80px] p-3 rounded-xl border ${stage.border} ${stage.bg} transition-all duration-300 hover:scale-105 hover:shadow-lg`}
   310→            >
   311→              <Icon className={`w-5 h-5 ${stage.textColor} mb-1`} />
   312→              <span className={`text-xl font-bold ${stage.textColor}`}>{count}</span>
   313→              <span className="text-[10px] text-slate-400 mt-0.5">{stage.label}</span>
   314→              {pct > 0 && (
   315→                <span className="text-[9px] text-slate-500 font-mono mt-0.5">{pct}%</span>
   316→              )}
   317→            </motion.div>
   318→            {i < PIPELINE_STAGES.length - 1 && (
   319→              <motion.div
   320→                initial={{ opacity: 0, x: -8 }}
   321→                animate={{ opacity: 1, x: 0 }}
   322→                transition={{ delay: i * 0.08 + 0.04, duration: 0.3 }}
   323→                className="flex items-center px-1"
   324→              >
   325→                <ChevronRight className="w-4 h-4 text-slate-600" />
   326→              </motion.div>
   327→            )}
   328→          </React.Fragment>
   329→        )
   330→      })}
   331→    </div>
   332→  )
   333→}
