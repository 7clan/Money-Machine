   221→   221→// ─── Animated Counter Hook ───────────────────────────────────────────
   222→   222→function useAnimatedCounter(end: number, duration: number = 800) {
   223→   223→  const [count, setCount] = useState(end)
   224→   224→  const prevEnd = useRef(end)
   225→   225→  useEffect(() => {
   226→   226→    if (prevEnd.current === end) return
   227→   227→    const start = prevEnd.current
   228→   228→    const startTime = performance.now()
   229→   229→    const step = (now: number) => {
   230→   230→      const progress = Math.min((now - startTime) / duration, 1)
   231→   231→      const eased = 1 - Math.pow(1 - progress, 3) // easeOutCubic
   232→   232→      setCount(Math.round(start + (end - start) * eased))
   233→   233→      if (progress < 1) requestAnimationFrame(step)
   234→   234→    }
   235→   235→    requestAnimationFrame(step)
   236→   236→    prevEnd.current = end
   237→   237→    return () => {}
   238→   238→  }, [end, duration])
   239→   239→  return count
   240→   240→}

   295→   295→function PipelineFlow({ pipeline }: { pipeline: AgentStatus['pipeline'] | null }) {
   296→   296→  if (!pipeline) return null
   297→   297→  const total = pipeline ? Object.values(pipeline).reduce((a: number, b: number) => a + b, 0) : 0
   298→   298→  return (
   299→   299→    <div className="flex items-center gap-0 overflow-x-auto py-2 px-1">
   300→   300→      {PIPELINE_STAGES.map((stage, i) => {
   301→   301→        const count = pipeline[stage.key as keyof typeof pipeline] || 0
   302→   302→        const pct = total > 0 ? Math.round((count / total) * 100) : 0
   303→   303→        const Icon = stage.icon
   304→   304→        return (
   305→   305→          <React.Fragment key={stage.key}>
   306→   306→            <motion.div
   307→   307→              initial={{ opacity: 0, scale: 0.8 }}
   308→   308→              animate={{ opacity: 1, scale: 1 }}
   309→   309→              transition={{ delay: i * 0.08, duration: 0.3 }}
   310→   310→              className={`flex flex-col items-center min-w-[80px] p-3 rounded-xl border ${stage.border} ${stage.bg} transition-all duration-300 hover:scale-105 hover:shadow-lg`}
   311→   311→            >
   312→   312→              <Icon className={`w-5 h-5 ${stage.textColor} mb-1`} />
   313→   313→              <span className={`text-xl font-bold ${stage.textColor}`}>{count}</span>
   314→   314→              <span className="text-[10px] text-slate-400 mt-0.5">{stage.label}</span>
   315→   315→              {pct > 0 && (
   316→   316→                <span className="text-[9px] text-slate-500 font-mono mt-0.5">{pct}%</span>

   318→   318→            </motion.div>

   320→   320→              <motion.div
   321→   321→                initial={{ opacity: 0, x: -8 }}
   322→   322→                animate={{ opacity: 1, x: 0 }}
   323→   323→                transition={{ delay: i * 0.08 + 0.04, duration: 0.3 }}
