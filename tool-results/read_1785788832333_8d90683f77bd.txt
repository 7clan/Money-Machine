   122→function stateColor(state: string) {
   123→  const activeStates = ['running', 'researching_niches', 'creating_strategy', 'researching_topic', 'writing_script', 'producing_video', 'reviewing', 'uploading', 'cycle_complete']
   124→  if (activeStates.includes(state)) return { dot: 'bg-emerald-500', ring: 'ring-emerald-500/30', shadow: 'shadow-emerald-500/20', text: 'text-emerald-400', label: 'Running' }
   125→  if (state === 'error') return { dot: 'bg-red-500', ring: 'ring-red-500/30', shadow: 'shadow-red-500/20', text: 'text-red-400', label: 'Error' }
   126→  if (state === 'paused') return { dot: 'bg-amber-500', ring: 'ring-amber-500/30', shadow: 'shadow-amber-500/20', text: 'text-amber-400', label: 'Paused' }
   127→  return { dot: 'bg-slate-400', ring: 'ring-slate-400/30', shadow: 'shadow-slate-400/20', text: 'text-slate-400', label: 'Idle' }
   128→}
   129→
   130→function actionColor(action: string) {
   131→  if (action.includes('error') || action.includes('fail')) return 'text-red-400 bg-red-500/10 border-red-500/30'
   132→  if (action.includes('upload') || action.includes('complete')) return 'text-emerald-400 bg-emerald-500/10 border-emerald-500/30'
   133→  if (action.includes('produce') || action.includes('render')) return 'text-cyan-400 bg-cyan-500/10 border-cyan-500/30'
   134→  if (action.includes('script') || action.includes('write')) return 'text-amber-400 bg-amber-500/10 border-amber-500/30'
   135→  if (action.includes('research') || action.includes('niche')) return 'text-blue-400 bg-blue-500/10 border-blue-500/30'
   136→  if (action.includes('review') || action.includes('quality')) return 'text-rose-400 bg-rose-500/10 border-rose-500/30'
   137→  return 'text-slate-400 bg-slate-500/10 border-slate-500/30'
   138→}

   301→  const [status, setStatus] = useState<AgentStatus | null>(null)

   308→  const [activeTab, setActiveTab] = useState('overview')
   309→  const [lastPoll, setLastPoll] = useState<Date>(new Date())
   310→
   311→  // ── Polling ─────────────────────────────────────────────────────
   312→  const fetchStatus = useCallback(async () => {
   313→    try {
   314→      const res = await fetch('/api/agent/status')
   315→      if (res.ok) setStatus(await res.json())
   316→    } catch {}
   317→  }, [])
   318→
   319→  const fetchPipeline = useCallback(async () => {
   320→    try {
   321→      const res = await fetch('/api/data/pipeline')
   322→      if (res.ok) setPipeline(await res.json())
   323→    } catch {}
   324→  }, [])
   325→
   326→  const fetchChannel = useCallback(async () => {
   327→    try {
   328→      const res = await fetch('/api/data/channel')
   329→      if (res.ok) setChannel(await res.json())
   330→    } catch {}
   331→  }, [])
   332→
   333→  const fetchJobs = useCallback(async () => {
   334→    try {
   335→      const res = await fetch('/api/data/jobs')
   336→      if (res.ok) setJobs(await res.json())
   337→    } catch {}
   338→  }, [])
   339→
   340→  const fetchLogs = useCallback(async () => {
   341→    try {
   342→      const res = await fetch('/api/data/audit-logs')
   343→      if (res.ok) setLogs(await res.json())
   344→    } catch {}
   345→  }, [])
   346→
   347→  const fetchAnalytics = useCallback(async () => {
   348→    try {
   349→      const res = await fetch('/api/data/analytics')
   350→      if (res.ok) setAnalytics(await res.json())
   351→    } catch {}
   352→  }, [])
   353→
   354→  const pollAll = useCallback(async () => {
   355→    await Promise.all([fetchStatus(), fetchPipeline(), fetchChannel(), fetchJobs(), fetchLogs(), fetchAnalytics()])
   356→    setLastPoll(new Date())
   357→  }, [fetchStatus, fetchPipeline, fetchChannel, fetchJobs, fetchLogs, fetchAnalytics])
   358→
   359→  useEffect(() => {
   360→    pollAll()
   361→    const interval = setInterval(pollAll, 5000)
   362→    return () => clearInterval(interval)
   363→  }, [pollAll])
   364→
   365→  // ── Commands ────────────────────────────────────────────────────
   366→  const sendCommand = async (command: string, extra?: any) => {
   367→    setLoading(true)
   368→    try {
   369→      const res = await fetch('/api/agent/command', {
   370→        method: 'POST',
   371→        headers: { 'Content-Type': 'application/json' },
   372→        body: JSON.stringify({ command, ...extra }),
   373→      })
   374→      const data = await res.json()
   375→      await fetchStatus()
   376→      return data
   377→    } catch (e: any) {
   378→      console.error(e)
   379→    } finally {
   380→      setLoading(false)
   381→    }
   382→  }
   383→
   384→  // ── Computed ────────────────────────────────────────────────────
   385→  const totalPipeline = status ? Object.values(status.pipeline).reduce((a: number, b: number) => a + b, 0) : 0

   393→      produced: Math.max(0, (status?.pipeline.producing || 0) + Math.floor(Math.random() * 2)),
   394→      uploaded: Math.max(0, (status?.pipeline.uploaded || 0) - Math.floor(Math.random() * i * 0.1)),
   395→    }))
   396→  }, [status?.pipeline, logs.length])
   397→
   398→  const nicheBarData = useMemo(() => {
   399→    if (!channel?.niches?.length) return []

   401→      name: n.nicheName.length > 18 ? n.nicheName.slice(0, 18) + '...' : n.nicheName,
   402→      score: n.compositeScore || 0,
   403→      selected: n.isSelected,
   404→    }))
   405→  }, [channel?.niches])
   406→
   407→  // ── Render ──────────────────────────────────────────────────────
   408→  return (
   409→    <div className="min-h-screen bg-slate-950 text-slate-100 flex flex-col">
   410→      {/* ═══ TOP BAR ═══ */}

   412→        <div className="px-4 md:px-6 py-3 flex items-center justify-between gap-3">
   413→          {/* Logo */}
   414→          <div className="flex items-center gap-3 shrink-0">
   415→            <div className="w-9 h-9 rounded-xl bg-gradient-to-br from-red-500 to-red-700 flex items-center justify-center shadow-lg shadow-red-500/20">
   416→              <Youtube className="w-5 h-5 text-white" />
   417→            </div>

   419→              <h1 className="text-base font-bold tracking-tight bg-gradient-to-r from-white to-slate-300 bg-clip-text text-transparent">
   420→                Revenue Studio
   421→              </h1>
   422→              <p className="text-[10px] text-slate-500 uppercase tracking-widest">Mission Control</p>
   423→            </div>
   424→          </div>
   425→
   426→          {/* Center: Agent State + Mode */}
   427→          <div className="flex items-center gap-3 flex-1 justify-center">
   428→            {status && <AgentStateIndicator state={status.state} />}
   429→            <Badge className={`text-[10px] ${MODES.find(m => m.key === status?.operatingMode)?.badge || 'bg-slate-500/20 text-slate-300'}`}>
   430→              {modeLabel(status?.operatingMode || 'private_production')}
   431→            </Badge>
   432→          </div>
   433→
   434→          {/* Right: Emergency Stop + YT Connection */}
   435→          <div className="flex items-center gap-2 shrink-0">
   436→            <Badge variant={channel?.youtubeConnected ? 'default' : 'outline'} className={`text-[10px] ${channel?.youtubeConnected ? 'bg-red-500/20 text-red-300 border-red-500/30' : 'border-slate-600 text-slate-400'}`}>
   437→              <Youtube className="w-3 h-3 mr-1" />
   438→              {channel?.youtubeConnected ? 'Connected' : 'Offline'}
   439→            </Badge>
   440→
   441→            {/* EMERGENCY STOP - ALWAYS VISIBLE */}
   442→            <motion.div whileTap={{ scale: 0.95 }}>
   443→              <Button
