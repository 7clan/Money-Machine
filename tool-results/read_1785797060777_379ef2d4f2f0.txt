   190→   190→function toTimestamp(value?: string | Date | null): number {
   191→   191→  if (!value) return 0
   192→   192→  const d = value instanceof Date ? value : new Date(value)
   193→   193→  const t = d.getTime()
   194→   194→  return Number.isNaN(t) ? 0 : t
   195→   195→}
   196→   196→
   197→   197→function relativeTime(value?: string | Date | null): string {
   198→   198→  const t = toTimestamp(value)
   199→   199→  if (!t) return '—'
   200→   200→  const diff = Date.now() - t
   201→   201→  if (diff < 0) return 'just now'
   202→   202→  const sec = Math.floor(diff / 1000)
   203→   203→  if (sec < 60) return `${sec}s ago`
   204→   204→  const min = Math.floor(sec / 60)
   205→   205→  if (min < 60) return `${min}m ago`
   206→   206→  const hr = Math.floor(min / 60)
   207→   207→  if (hr < 24) return `${hr}h ago`
   208→   208→  const day = Math.floor(hr / 24)
   209→   209→  if (day < 7) return `${day}d ago`
   210→   210→  const week = Math.floor(day / 7)
   211→   211→  if (week < 4) return `${week}w ago`
   212→   212→  const month = Math.floor(day / 30)
   213→   213→  if (month < 12) return `${month}mo ago`
   214→   214→  const year = Math.floor(day / 365)
   215→   215→  return `${year}y ago`
   216→   216→}
   217→   217→
   218→   218→function formatDate(value?: string | Date | null): string {
   219→   219→  const t = toTimestamp(value)
   220→   220→  if (!t) return '—'
   221→   221→  return new Date(t).toLocaleDateString(undefined, {
   222→   222→    year: 'numeric',
   223→   223→    month: 'short',
   224→   224→    day: 'numeric',
   225→   225→    hour: '2-digit',
   226→   226→    minute: '2-digit',
   227→   227→  })
   228→   228→}
   229→   229→
   230→   230→function clampScore(value?: number | null): number {
   231→   231→  if (value === null || value === undefined || Number.isNaN(value)) return 0
   232→   232→  if (value < 0) return 0
   233→   233→  if (value > 100) return 100
   234→   234→  return Math.round(value)
   235→   235→}

   237→   237→function truncate(text: string, max: number): string {
   238→   238→  if (text.length <= max) return text
   239→   239→  return text.slice(0, max - 1).trimEnd() + '…'
   240→   240→}

   242→   242→function scoreBarColor(metric: string, value: number): string {
   243→   243→  // Higher-is-better metrics
   244→   244→  const higherBetter = [
   245→   245→    'originalityScore',
   246→   246→    'retentionPrediction',
   247→   247→    'revenuePotential',
   248→   248→    'compositeScore',
   249→   249→    'searchVolume',
   250→   250→  ]
   251→   251→  // Lower-is-better metrics
   252→   252→  const lowerBetter = ['competitionScore', 'productionEffort', 'riskScore']
   253→   253→  if (higherBetter.includes(metric)) {
   254→   254→    if (value >= 75) return 'bg-gradient-to-r from-violet-500 to-cyan-500'
   255→   255→    if (value >= 50) return 'bg-gradient-to-r from-emerald-500 to-cyan-500'
   256→   256→    if (value >= 30) return 'bg-amber-500'
   257→   257→    return 'bg-rose-500'
   258→   258→  }
   259→   259→  if (lowerBetter.includes(metric)) {
   260→   260→    if (value <= 30) return 'bg-gradient-to-r from-emerald-500 to-cyan-500'
   261→   261→    if (value <= 50) return 'bg-amber-500'
   262→   262→    return 'bg-rose-500'
   263→   263→  }
   264→   264→  return 'bg-primary'
   265→   265→}

   267→   267→function pillarColor(color?: string | null): string {
   268→   268→  if (!color) return 'bg-slate-400'
   269→   269→  // Map known hex / named colors to closest tailwind-ish dot color
   270→   270→  const c = color.toLowerCase()
   271→   271→  if (c.includes('violet') || c.includes('purple')) return 'bg-violet-400'
   272→   272→  if (c.includes('cyan') || c.includes('teal')) return 'bg-cyan-400'
   273→   273→  if (c.includes('emerald') || c.includes('green')) return 'bg-emerald-400'
   274→   274→  if (c.includes('amber') || c.includes('orange') || c.includes('yellow'))
   275→   275→    return 'bg-amber-400'
   276→   276→  if (c.includes('rose') || c.includes('pink') || c.includes('red'))
   277→   277→    return 'bg-rose-400'
   278→   278→  if (c.includes('#')) {
   279→   279→    // try to use the raw color via inline style fallback in component
   280→   280→    return ''
   281→   281→  }
   282→   282→  return 'bg-slate-400'
   283→   283→}

   287→   287→export function IdeaExplorer({
   288→   288→  ideas,
   289→   289→  onSelectIdea,
   290→   290→  className,
   291→   291→}: IdeaExplorerProps) {
   292→   292→  const [searchInput, setSearchInput] = useState('')
   293→   293→  const [debouncedSearch, setDebouncedSearch] = useState('')
   294→   294→  const [typeFilter, setTypeFilter] = useState<'all' | IdeaType>('all')
   295→   295→  const [statusFilter, setStatusFilter] = useState<'all' | IdeaStatus>('all')
   296→   296→  const [pillarFilter, setPillarFilter] = useState<'all' | string>('all')
   297→   297→  const [sortKey, setSortKey] = useState<SortKey>('created_desc')
   298→   298→  const [drawerIdeaId, setDrawerIdeaId] = useState<string | null>(null)
   299→   299→
   300→   300→  // Debounced search (200ms)
   301→   301→  useEffect(() => {
   302→   302→    const handle = setTimeout(() => {
   303→   303→      setDebouncedSearch(searchInput.trim().toLowerCase())
   304→   304→    }, 200)
   305→   305→    return () => clearTimeout(handle)
   306→   306→  }, [searchInput])
   307→   307→
   308→   308→  // Reset visible count when filters change.
   309→   309→  // Pattern: store a signature alongside the count; when the signature
   310→   310→  // changes during render, reset the count to the baseline. This is the
   311→   311→  // React-recommended "derive state during render" pattern and avoids
   312→   312→  // both the setState-in-effect and ref-during-render lint rules.
   313→   313→  const filterSignature = `${debouncedSearch}|${typeFilter}|${statusFilter}|${pillarFilter}|${sortKey}`
   314→   314→  const [visibleState, setVisibleState] = useState<{
   315→   315→    sig: string
   316→   316→    count: number
   317→   317→  }>({ sig: filterSignature, count: VISIBLE_INCREMENT })
   318→   318→  if (visibleState.sig !== filterSignature) {
   319→   319→    setVisibleState({ sig: filterSignature, count: VISIBLE_INCREMENT })
   320→   320→  }
   321→   321→  const visibleCount = visibleState.count
   322→   322→
   323→   323→  // Derive pillar list from ideas
   324→   324→  const pillars = useMemo(() => {
   325→   325→    const map = new Map<string, { id: string; name: string }>()
   326→   326→    for (const idea of ideas) {
   327→   327→      const p = idea.pillar
   328→   328→      if (p?.id && p?.name) {
   329→   329→        if (!map.has(p.id)) {
   330→   330→          map.set(p.id, { id: p.id, name: p.name })
   331→   331→        }
   332→   332→      } else if (idea.pillarId) {
   333→   333→        if (!map.has(idea.pillarId)) {
   334→   334→          map.set(idea.pillarId, { id: idea.pillarId, name: idea.pillarId })
   335→   335→        }
   336→   336→      }
   337→   337→    }
   338→   338→    return Array.from(map.values()).sort((a, b) => a.name.localeCompare(b.name))
   339→   339→  }, [ideas])
   340→   340→
   341→   341→  const activeFilterCount = useMemo(() => {
   342→   342→    let n = 0
   343→   343→    if (debouncedSearch) n++
   344→   344→    if (typeFilter !== 'all') n++
   345→   345→    if (statusFilter !== 'all') n++
   346→   346→    if (pillarFilter !== 'all') n++
   347→   347→    return n
   348→   348→  }, [debouncedSearch, typeFilter, statusFilter, pillarFilter])
   349→   349→
   350→   350→  const filteredIdeas = useMemo(() => {
   351→   351→    const out: Idea[] = []
   352→   352→    for (const idea of ideas) {
   353→   353→      if (
   354→   354→        debouncedSearch &&
   355→   355→        !idea.title.toLowerCase().includes(debouncedSearch)
   356→   356→      ) {
   357→   357→        continue
   358→   358→      }
   359→   359→      if (typeFilter !== 'all' && idea.type !== typeFilter) continue
   360→   360→      if (statusFilter !== 'all' && idea.status !== statusFilter) continue
   361→   361→      if (pillarFilter !== 'all') {
   362→   362→        const pid = idea.pillar?.id ?? idea.pillarId ?? null
   363→   363→        if (pid !== pillarFilter) continue
   364→   364→      }
   365→   365→      out.push(idea)
   366→   366→    }
   367→   367→
   368→   368→    out.sort((a, b) => {
   369→   369→      switch (sortKey) {
   370→   370→        case 'created_asc':
   371→   371→          return toTimestamp(a.createdAt) - toTimestamp(b.createdAt)
   372→   372→        case 'created_desc':
   373→   373→          return toTimestamp(b.createdAt) - toTimestamp(a.createdAt)
   374→   374→        case 'title_asc':
   375→   375→          return a.title.localeCompare(b.title)
   376→   376→        case 'title_desc':
   377→   377→          return b.title.localeCompare(a.title)
   378→   378→        case 'composite_desc':
   379→   379→          return (
   380→   380→            (b.compositeScore ?? -1) - (a.compositeScore ?? -1)
   381→   381→          )
   382→   382→        case 'scheduled_asc': {
   383→   383→          const ta = toTimestamp(a.scheduledDate)
   384→   384→          const tb = toTimestamp(b.scheduledDate)
   385→   385→          // unscheduled go to end
   386→   386→          if (!ta && !tb) return 0
   387→   387→          if (!ta) return 1
   388→   388→          if (!tb) return -1
   389→   389→          return ta - tb
   390→   390→        }
   391→   391→        default:
   392→   392→          return 0
   393→   393→      }
   394→   394→    })
   395→   395→    return out
   396→   396→  }, [
   397→   397→    ideas,
   398→   398→    debouncedSearch,
   399→   399→    typeFilter,
   400→   400→    statusFilter,
   401→   401→    pillarFilter,
   402→   402→    sortKey,
   403→   403→  ])
   404→   404→
   405→   405→  const visibleIdeas = filteredIdeas.slice(0, visibleCount)
   406→   406→  const hasMore = filteredIdeas.length > visibleCount
   407→   407→
   408→   408→  const selectedIdea = useMemo(() => {
   409→   409→    if (!drawerIdeaId) return null
   410→   410→    return ideas.find((i) => i.id === drawerIdeaId) ?? null
   411→   411→  }, [drawerIdeaId, ideas])
   412→   412→
   413→   413→  const clearFilters = useCallback(() => {
   414→   414→    setSearchInput('')
   415→   415→    setDebouncedSearch('')
   416→   416→    setTypeFilter('all')
   417→   417→    setStatusFilter('all')
   418→   418→    setPillarFilter('all')
   419→   419→    setSortKey('created_desc')
   420→   420→  }, [])
   421→   421→
   422→   422→  const handleCardClick = useCallback((ideaId: string) => {
   423→   423→    setDrawerIdeaId(ideaId)
   424→   424→  }, [])
   425→   425→
   426→   426→  const handleSelectFromDrawer = useCallback(() => {
   427→   427→    if (drawerIdeaId && onSelectIdea) {
   428→   428→      onSelectIdea(drawerIdeaId)
   429→   429→    }
   430→   430→  }, [drawerIdeaId, onSelectIdea])
   431→   431→
   432→   432→  const isLoading = ideas.length === 0
   433→   433→
   434→   434→  // ─── Render ────────────────────────────────────────────────────────
   435→   435→
   436→   436→  return (
   437→   437→    <div
   438→   438→      className={cn(
   439→   439→        'w-full rounded-xl bg-slate-950 p-3 sm:p-4',
   440→   440→        'border border-slate-800/50',
   441→   441→        className,
   442→   442→      )}
   443→   443→    >
   444→   444→      {/* ─── Header / Search ─── */}
   445→   445→      <div className="mb-3 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   446→   446→        <div className="flex items-center gap-2">
   447→   447→          <div className="flex size-8 items-center justify-center rounded-lg bg-gradient-to-br from-violet-500/20 to-cyan-500/20 border border-violet-500/30">
   448→   448→            <Lightbulb className="size-4 text-violet-300" />
   449→   449→          </div>
   450→   450→          <div>
   451→   451→            <h3 className="text-sm font-semibold text-slate-100">
   452→   452→              Idea Explorer
   453→   453→            </h3>
   454→   454→            <p className="text-[11px] text-slate-400">
   455→   455→              Search, filter &amp; inspect video ideas
   456→   456→            </p>
   457→   457→          </div>
   458→   458→        </div>
   459→   459→
   460→   460→        <div className="relative w-full sm:w-80">
   461→   461→          <Search className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-slate-500" />
   462→   462→          <Input
   463→   463→            value={searchInput}
   464→   464→            onChange={(e) => setSearchInput(e.target.value)}
   465→   465→            placeholder="Search ideas by title…"
   466→   466→            className="h-9 border-slate-700/60 bg-slate-900/60 pl-9 pr-8 text-sm text-slate-100 placeholder:text-slate-500 focus-visible:border-violet-500/50 focus-visible:ring-violet-500/20"
   467→   467→          />
   468→   468→          {searchInput && (
   469→   469→            <button
   470→   470→              type="button"
   471→   471→              onClick={() => setSearchInput('')}
   472→   472→              className="absolute right-2 top-1/2 -translate-y-1/2 rounded p-1 text-slate-500 hover:bg-slate-800 hover:text-slate-300"
   473→   473→              aria-label="Clear search"
   474→   474→            >
   475→   475→              <X className="size-3.5" />
   476→   476→            </button>
   477→   477→          )}
   478→   478→        </div>
   479→   479→      </div>
