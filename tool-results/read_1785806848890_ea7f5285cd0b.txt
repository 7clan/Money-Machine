   170→function stateColor(state: string) {

   172→  if (activeStates.includes(state)) return { dot: 'bg-emerald-500', ring: 'ring-emerald-500/30', shadow: 'shadow-emerald-500/20', text: 'text-emerald-400', label: 'Running' }
   173→  if (state === 'error') return { dot: 'bg-red-500', ring: 'ring-red-500/30', shadow: 'shadow-red-500/20', text: 'text-red-400', label: 'Error' }
   174→  if (state === 'paused') return { dot: 'bg-amber-500', ring: 'ring-amber-500/30', shadow: 'shadow-amber-500/20', text: 'text-amber-400', label: 'Paused' }
   175→  return { dot: 'bg-slate-400', ring: 'ring-slate-400/30', shadow: 'shadow-slate-400/20', text: 'text-slate-400', label: 'Idle' }
   176→}

   178→function actionColor(action: string) {
   179→  // New categorized actions
   180→  if (action === 'emergency_stop') return 'text-red-400 bg-red-500/10 border-red-500/30'
   181→  if (action === 'mode_change') return 'text-amber-400 bg-amber-500/10 border-amber-500/30'
   182→  if (action === 'upload') return 'text-cyan-400 bg-cyan-500/10 border-cyan-500/30'
   183→  if (action === 'strategy_change') return 'text-violet-400 bg-violet-500/10 border-violet-500/30'
   184→  // Legacy / fallback keyword matching
   185→  if (action.includes('error') || action.includes('fail')) return 'text-red-400 bg-red-500/10 border-red-500/30'
   186→  if (action.includes('upload') || action.includes('complete')) return 'text-emerald-400 bg-emerald-500/10 border-emerald-500/30'
   187→  if (action.includes('produce') || action.includes('render')) return 'text-cyan-400 bg-cyan-500/10 border-cyan-500/30'
   188→  if (action.includes('script') || action.includes('write')) return 'text-amber-400 bg-amber-500/10 border-amber-500/30'
   189→  if (action.includes('research') || action.includes('niche')) return 'text-cyan-400 bg-cyan-500/10 border-cyan-500/30'
   190→  if (action.includes('review') || action.includes('quality')) return 'text-rose-400 bg-rose-500/10 border-rose-500/30'
   191→  return 'text-slate-400 bg-slate-500/10 border-slate-500/30'

   221→// ─── Animated Counter Hook ───────────────────────────────────────────
   222→function useAnimatedCounter(end: number, duration: number = 800) {
   223→  const [count, setCount] = useState(end)
   224→  const prevEnd = useRef(end)
   225→  useEffect(() => {
   226→    if (prevEnd.current === end) return
   227→    const start = prevEnd.current
   228→    const startTime = performance.now()
   229→    const step = (now: number) => {
   230→      const progress = Math.min((now - startTime) / duration, 1)
   231→      const eased = 1 - Math.pow(1 - progress, 3) // easeOutCubic
   232→      setCount(Math.round(start + (end - start) * eased))
   233→      if (progress < 1) requestAnimationFrame(step)
   234→    }
   235→    requestAnimationFrame(step)
   236→    prevEnd.current = end
   237→    return () => {}
   238→  }, [end, duration])
   239→  return count
   240→}

   261→function StatusCard({ icon: Icon, label, value, sub, trend, color = 'text-emerald-400', valueSuffix, hint }: {
   262→  icon: any; label: string; value: string | number; sub?: string; trend?: 'up' | 'down' | 'flat'; color?: string; valueSuffix?: string; hint?: string
   263→}) {
   264→  const numericValue = typeof value === 'number' ? value : 0
   265→  const animatedValue = useAnimatedCounter(numericValue)
   266→  const displayValue = typeof value === 'number' ? animatedValue : value
   267→  const bgForColor = (c: string) => c === 'text-emerald-400' ? 'bg-emerald-500/10' : c === 'text-blue-400' ? 'bg-blue-500/10' : c === 'text-amber-400' ? 'bg-amber-500/10' : c === 'text-violet-400' ? 'bg-violet-500/10' : c === 'text-cyan-400' ? 'bg-cyan-500/10' : c === 'text-rose-400' ? 'bg-rose-500/10' : 'bg-slate-500/10'
   268→  const trendConfig = trend === 'up' ? { icon: ArrowUpRight, color: 'text-emerald-400', bg: 'bg-emerald-500/20' } : trend === 'down' ? { icon: ArrowDownRight, color: 'text-red-400', bg: 'bg-red-500/20' } : trend === 'flat' ? { icon: Minus, color: 'text-slate-400', bg: 'bg-slate-500/20' } : null
   269→  return (
   270→    <GradientCard>
   271→      <div className="p-4 group cursor-default" title={hint}>
   272→        <div className="flex items-start justify-between">
   273→          <div className={`p-2.5 rounded-xl ${bgForColor(color)} transition-all duration-300 group-hover:scale-110 group-hover:shadow-lg`}>
   274→            <Icon className={`w-5 h-5 ${color}`} />
   275→          </div>
   276→          {trendConfig && (
   277→            <div className={`flex items-center gap-0.5 text-xs px-1.5 py-0.5 rounded-full ${trendConfig.bg} ${trendConfig.color}`}>
   278→              <trendConfig.icon className="w-3 h-3" />
   279→            </div>
   280→          )}
   281→        </div>
   282→        <div className="mt-3">
   283→          <div className="flex items-baseline gap-1">
   284→            <p className="text-2xl sm:text-3xl font-bold tracking-tight truncate" title={String(value)}>{displayValue}</p>
   285→            {valueSuffix && <span className={`text-sm font-semibold ${color}`}>{valueSuffix}</span>}
   286→          </div>
   287→          <p className="text-xs text-slate-400 mt-1 font-medium">{label}</p>
   288→          {sub && <p className="text-[10px] text-slate-500 mt-0.5 truncate" title={sub}>{sub}</p>}
   289→        </div>
   290→      </div>
   291→    </GradientCard>
   292→  )

   295→function PipelineFlow({ pipeline }: { pipeline: AgentStatus['pipeline'] | null }) {
   296→  if (!pipeline) return null
   297→  const total = pipeline ? Object.values(pipeline).reduce((a: number, b: number) => a + b, 0) : 0
   298→  return (
   299→    <div className="flex items-center gap-0 overflow-x-auto py-2 px-1">
   300→      {PIPELINE_STAGES.map((stage, i) => {
   301→        const count = pipeline[stage.key as keyof typeof pipeline] || 0
   302→        const pct = total > 0 ? Math.round((count / total) * 100) : 0
   303→        const Icon = stage.icon
   304→        return (
   305→          <React.Fragment key={stage.key}>
   306→            <motion.div
   307→              initial={{ opacity: 0, scale: 0.8 }}
   308→              animate={{ opacity: 1, scale: 1 }}
   309→              transition={{ delay: i * 0.08, duration: 0.3 }}
   310→              className={`flex flex-col items-center min-w-[80px] p-3 rounded-xl border ${stage.border} ${stage.bg} transition-all duration-300 hover:scale-105 hover:shadow-lg`}
   311→            >
   312→              <Icon className={`w-5 h-5 ${stage.textColor} mb-1`} />
   313→              <span className={`text-xl font-bold ${stage.textColor}`}>{count}</span>
   314→              <span className="text-[10px] text-slate-400 mt-0.5">{stage.label}</span>
   315→              {pct > 0 && (
   316→                <span className="text-[9px] text-slate-500 font-mono mt-0.5">{pct}%</span>
   317→              )}
   318→            </motion.div>
   319→            {i < PIPELINE_STAGES.length - 1 && (
   320→              <motion.div
   321→                initial={{ opacity: 0, x: -8 }}
   322→                animate={{ opacity: 1, x: 0 }}
   323→                transition={{ delay: i * 0.08 + 0.04, duration: 0.3 }}
   324→                className="flex items-center px-1"
   325→              >
   326→                <ChevronRight className="w-4 h-4 text-slate-600" />
   327→              </motion.div>
   328→            )}
   329→          </React.Fragment>
   330→        )
   331→      })}
   332→    </div>
   333→  )
   334→}
