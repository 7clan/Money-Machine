   157→function formatDuration(seconds?: number | null): string {
   158→  if (seconds == null || Number.isNaN(seconds)) return '—'
   159→  const s = Math.round(seconds)
   160→  const m = Math.floor(s / 60)
   161→  const r = s % 60
   162→  return `${m}:${r.toString().padStart(2, '0')}`
   163→}
   164→
   165→function formatFileSize(bytes?: number | null): string {
   166→  if (!bytes) return '—'
   167→  if (bytes < 1024) return `${bytes} B`
   168→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   169→  if (bytes < 1024 * 1024 * 1024) return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   170→  return `${(bytes / (1024 * 1024 * 1024)).toFixed(2)} GB`
   171→}

   173→// Map pillar.color (any CSS color) into a safe tailwind text/bg class set.
   174→// Falls back to the violet→cyan brand gradient.
   175→function pillarStyle(color?: string | null): { text: string; bg: string; border: string } {
   176→  const c = (color || '').toLowerCase()
   177→  switch (c) {
   178→    case 'violet':
   179→    case '#8b5cf6':
   180→    case 'purple':
   181→      return { text: 'text-violet-300', bg: 'bg-violet-500/10', border: 'border-violet-500/30' }
   182→    case 'cyan':
   183→    case '#06b6d4':
   184→      return { text: 'text-cyan-300', bg: 'bg-cyan-500/10', border: 'border-cyan-500/30' }
   185→    case 'emerald':
   186→    case '#10b981':
   187→    case 'green':
   188→      return { text: 'text-emerald-300', bg: 'bg-emerald-500/10', border: 'border-emerald-500/30' }
   189→    case 'amber':
   190→    case '#f59e0b':
   191→    case 'yellow':
   192→      return { text: 'text-amber-300', bg: 'bg-amber-500/10', border: 'border-amber-500/30' }
   193→    case 'rose':
   194→    case '#f43f5e':
   195→    case 'red':
   196→      return { text: 'text-rose-300', bg: 'bg-rose-500/10', border: 'border-rose-500/30' }
   197→    default:
   198→      return { text: 'text-violet-300', bg: 'bg-violet-500/10', border: 'border-violet-500/30' }
   199→  }
   200→}

   202→// Color-code VideoProject.status.
   203→function statusStyle(status: string): { label: string; text: string; bg: string; border: string; dot: string } {
   204→  const s = (status || '').toLowerCase()
   205→  if (['approved', 'uploaded', 'published', 'completed'].includes(s)) {
   206→    return { label: status, text: 'text-emerald-300', bg: 'bg-emerald-500/10', border: 'border-emerald-500/30', dot: 'bg-emerald-400' }
   207→  }
   208→  if (['failed', 'rejected', 'error'].includes(s)) {
   209→    return { label: status, text: 'text-rose-300', bg: 'bg-rose-500/10', border: 'border-rose-500/30', dot: 'bg-rose-400' }
   210→  }
   211→  if (['rendering', 'editing', 'recording', 'collecting_assets', 'storyboarding', 'planning', 'uploading'].includes(s)) {
   212→    return { label: status, text: 'text-cyan-300', bg: 'bg-cyan-500/10', border: 'border-cyan-500/30', dot: 'bg-cyan-400' }
   213→  }
   214→  if (['review', 'reviewing'].includes(s)) {
   215→    return { label: status, text: 'text-amber-300', bg: 'bg-amber-500/10', border: 'border-amber-500/30', dot: 'bg-amber-400' }
   216→  }
   217→  return { label: status, text: 'text-slate-300', bg: 'bg-slate-500/10', border: 'border-slate-500/30', dot: 'bg-slate-400' }
   218→}

   220→// Parse a JSON-string array (issues, etc.) safely.
   221→function parseJsonArray(raw?: string | null): string[] {
   222→  if (!raw) return []
   223→  try {
   224→    const v = JSON.parse(raw)
   225→    if (Array.isArray(v)) return v.map((x) => String(x))
   226→    return [String(v)]
   227→  } catch {
   228→    return raw.split('\n').map((s) => s.trim()).filter(Boolean)
   229→  }
   230→}

   593→      {/* Issues */}
