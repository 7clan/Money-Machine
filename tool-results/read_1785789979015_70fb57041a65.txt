   190→function toTimestamp(value?: string | Date | null): number {
   191→  if (!value) return 0
   192→  const d = value instanceof Date ? value : new Date(value)
   193→  const t = d.getTime()
   194→  return Number.isNaN(t) ? 0 : t
   195→}
   196→
   197→function relativeTime(value?: string | Date | null): string {
   198→  const t = toTimestamp(value)
   199→  if (!t) return '—'
   200→  const diff = Date.now() - t
   201→  if (diff < 0) return 'just now'
   202→  const sec = Math.floor(diff / 1000)
   203→  if (sec < 60) return `${sec}s ago`
   204→  const min = Math.floor(sec / 60)
   205→  if (min < 60) return `${min}m ago`
   206→  const hr = Math.floor(min / 60)
   207→  if (hr < 24) return `${hr}h ago`
   208→  const day = Math.floor(hr / 24)
   209→  if (day < 7) return `${day}d ago`
   210→  const week = Math.floor(day / 7)
   211→  if (week < 4) return `${week}w ago`
   212→  const month = Math.floor(day / 30)
   213→  if (month < 12) return `${month}mo ago`
   214→  const year = Math.floor(day / 365)
   215→  return `${year}y ago`
   216→}
   217→
   218→function formatDate(value?: string | Date | null): string {
   219→  const t = toTimestamp(value)
   220→  if (!t) return '—'
   221→  return new Date(t).toLocaleDateString(undefined, {
   222→    year: 'numeric',
   223→    month: 'short',
   224→    day: 'numeric',
   225→    hour: '2-digit',
   226→    minute: '2-digit',
   227→  })
   228→}
   229→
   230→function clampScore(value?: number | null): number {
   231→  if (value === null || value === undefined || Number.isNaN(value)) return 0
   232→  if (value < 0) return 0
   233→  if (value > 100) return 100
   234→  return Math.round(value)
   235→}
   236→
   237→function truncate(text: string, max: number): string {
   238→  if (text.length <= max) return text
   239→  return text.slice(0, max - 1).trimEnd() + '…'
   240→}

   242→function scoreBarColor(metric: string, value: number): string {
   243→  // Higher-is-better metrics
   244→  const higherBetter = [
   245→    'originalityScore',
   246→    'retentionPrediction',
   247→    'revenuePotential',
   248→    'compositeScore',
   249→    'searchVolume',
   250→  ]
   251→  // Lower-is-better metrics
   252→  const lowerBetter = ['competitionScore', 'productionEffort', 'riskScore']
   253→  if (higherBetter.includes(metric)) {
   254→    if (value >= 75) return 'bg-gradient-to-r from-violet-500 to-cyan-500'
   255→    if (value >= 50) return 'bg-gradient-to-r from-emerald-500 to-cyan-500'
   256→    if (value >= 30) return 'bg-amber-500'
   257→    return 'bg-rose-500'
   258→  }
   259→  if (lowerBetter.includes(metric)) {
   260→    if (value <= 30) return 'bg-gradient-to-r from-emerald-500 to-cyan-500'
   261→    if (value <= 50) return 'bg-amber-500'
   262→    return 'bg-rose-500'
   263→  }
   264→  return 'bg-primary'
   265→}
   266→
   267→function pillarColor(color?: string | null): string {
   268→  if (!color) return 'bg-slate-400'
   269→  // Map known hex / named colors to closest tailwind-ish dot color
   270→  const c = color.toLowerCase()
   271→  if (c.includes('violet') || c.includes('purple')) return 'bg-violet-400'
   272→  if (c.includes('cyan') || c.includes('teal')) return 'bg-cyan-400'
   273→  if (c.includes('emerald') || c.includes('green')) return 'bg-emerald-400'
   274→  if (c.includes('amber') || c.includes('orange') || c.includes('yellow'))
   275→    return 'bg-amber-400'
   276→  if (c.includes('rose') || c.includes('pink') || c.includes('red'))
   277→    return 'bg-rose-400'
   278→  if (c.includes('#')) {
   279→    // try to use the raw color via inline style fallback in component
   280→    return ''
   281→  }
   282→  return 'bg-slate-400'
   283→}

   287→export function IdeaExplorer({
   288→  ideas,
   289→  onSelectIdea,
   290→  className,
   291→}: IdeaExplorerProps) {
   292→  const [searchInput, setSearchInput] = useState('')
   293→  const [debouncedSearch, setDebouncedSearch] = useState('')
   294→  const [typeFilter, setTypeFilter] = useState<'all' | IdeaType>('all')
   295→  const [statusFilter, setStatusFilter] = useState<'all' | IdeaStatus>('all')
   296→  const [pillarFilter, setPillarFilter] = useState<'all' | string>('all')
   297→  const [sortKey, setSortKey] = useState<SortKey>('created_desc')
   298→  const [drawerIdeaId, setDrawerIdeaId] = useState<string | null>(null)
   299→
   300→  // Debounced search (200ms)
   301→  useEffect(() => {
   302→    const handle = setTimeout(() => {
   303→      setDebouncedSearch(searchInput.trim().toLowerCase())
   304→    }, 200)
   305→    return () => clearTimeout(handle)
   306→  }, [searchInput])
   307→
   308→  // Reset visible count when filters change.
   309→  // Pattern: store a signature alongside the count; when the signature
   310→  // changes during render, reset the count to the baseline. This is the
   311→  // React-recommended "derive state during render" pattern and avoids
   312→  // both the setState-in-effect and ref-during-render lint rules.
   313→  const filterSignature = `${debouncedSearch}|${typeFilter}|${statusFilter}|${pillarFilter}|${sortKey}`
   314→  const [visibleState, setVisibleState] = useState<{
   315→    sig: string
   316→    count: number
   317→  }>({ sig: filterSignature, count: VISIBLE_INCREMENT })
   318→  if (visibleState.sig !== filterSignature) {
   319→    setVisibleState({ sig: filterSignature, count: VISIBLE_INCREMENT })
   320→  }
   321→  const visibleCount = visibleState.count
   322→
   323→  // Derive pillar list from ideas
   324→  const pillars = useMemo(() => {
   325→    const map = new Map<string, { id: string; name: string }>()
   326→    for (const idea of ideas) {
   327→      const p = idea.pillar
   328→      if (p?.id && p?.name) {
   329→        if (!map.has(p.id)) {
   330→          map.set(p.id, { id: p.id, name: p.name })
   331→        }
   332→      } else if (idea.pillarId) {
   333→        if (!map.has(idea.pillarId)) {
   334→          map.set(idea.pillarId, { id: idea.pillarId, name: idea.pillarId })
   335→        }
   336→      }
   337→    }
   338→    return Array.from(map.values()).sort((a, b) => a.name.localeCompare(b.name))
   339→  }, [ideas])
   340→
   341→  const activeFilterCount = useMemo(() => {
   342→    let n = 0
   343→    if (debouncedSearch) n++
   344→    if (typeFilter !== 'all') n++
   345→    if (statusFilter !== 'all') n++
   346→    if (pillarFilter !== 'all') n++
   347→    return n
   348→  }, [debouncedSearch, typeFilter, statusFilter, pillarFilter])
   349→
   350→  const filteredIdeas = useMemo(() => {
   351→    const out: Idea[] = []
   352→    for (const idea of ideas) {
   353→      if (
   354→        debouncedSearch &&
   355→        !idea.title.toLowerCase().includes(debouncedSearch)
   356→      ) {
   357→        continue
   358→      }
   359→      if (typeFilter !== 'all' && idea.type !== typeFilter) continue
   360→      if (statusFilter !== 'all' && idea.status !== statusFilter) continue
   361→      if (pillarFilter !== 'all') {
   362→        const pid = idea.pillar?.id ?? idea.pillarId ?? null
   363→        if (pid !== pillarFilter) continue
   364→      }
   365→      out.push(idea)
   366→    }
   367→
   368→    out.sort((a, b) => {
   369→      switch (sortKey) {
   370→        case 'created_asc':
   371→          return toTimestamp(a.createdAt) - toTimestamp(b.createdAt)
   372→        case 'created_desc':
   373→          return toTimestamp(b.createdAt) - toTimestamp(a.createdAt)
   374→        case 'title_asc':
   375→          return a.title.localeCompare(b.title)
   376→        case 'title_desc':
   377→          return b.title.localeCompare(a.title)
   378→        case 'composite_desc':
   379→          return (
   380→            (b.compositeScore ?? -1) - (a.compositeScore ?? -1)
   381→          )
   382→        case 'scheduled_asc': {
   383→          const ta = toTimestamp(a.scheduledDate)
   384→          const tb = toTimestamp(b.scheduledDate)
   385→          // unscheduled go to end
   386→          if (!ta && !tb) return 0
   387→          if (!ta) return 1
   388→          if (!tb) return -1
   389→          return ta - tb
   390→        }
   391→        default:
   392→          return 0
   393→      }
   394→    })
   395→    return out
   396→  }, [
   397→    ideas,
   398→    debouncedSearch,
   399→    typeFilter,
   400→    statusFilter,
   401→    pillarFilter,
   402→    sortKey,
   403→  ])
   404→
   405→  const visibleIdeas = filteredIdeas.slice(0, visibleCount)
   406→  const hasMore = filteredIdeas.length > visibleCount
   407→
   408→  const selectedIdea = useMemo(() => {
   409→    if (!drawerIdeaId) return null
   410→    return ideas.find((i) => i.id === drawerIdeaId) ?? null
   411→  }, [drawerIdeaId, ideas])
   412→
   413→  const clearFilters = useCallback(() => {
   414→    setSearchInput('')
   415→    setDebouncedSearch('')
   416→    setTypeFilter('all')
   417→    setStatusFilter('all')
   418→    setPillarFilter('all')
   419→    setSortKey('created_desc')
   420→  }, [])
   421→
   422→  const handleCardClick = useCallback((ideaId: string) => {
   423→    setDrawerIdeaId(ideaId)
   424→  }, [])
   425→
   426→  const handleSelectFromDrawer = useCallback(() => {
   427→    if (drawerIdeaId && onSelectIdea) {
   428→      onSelectIdea(drawerIdeaId)
   429→    }
   430→  }, [drawerIdeaId, onSelectIdea])
   431→
   432→  const isLoading = ideas.length === 0
   433→
   434→  // ─── Render ────────────────────────────────────────────────────────
   435→
   436→  return (
   437→    <div
   438→      className={cn(
   439→        'w-full rounded-xl bg-slate-950 p-3 sm:p-4',
   440→        'border border-slate-800/50',
   441→        className,
   442→      )}
   443→    >
   444→      {/* ─── Header / Search ─── */}
   445→      <div className="mb-3 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   446→        <div className="flex items-center gap-2">
   447→          <div className="flex size-8 items-center justify-center rounded-lg bg-gradient-to-br from-violet-500/20 to-cyan-500/20 border border-violet-500/30">
   448→            <Lightbulb className="size-4 text-violet-300" />
   449→          </div>
   450→          <div>
   451→            <h3 className="text-sm font-semibold text-slate-100">
   452→              Idea Explorer
   453→            </h3>
   454→            <p className="text-[11px] text-slate-400">
   455→              Search, filter &amp; inspect video ideas
   456→            </p>
   457→          </div>
   458→        </div>
   459→
   460→        <div className="relative w-full sm:w-80">
   461→          <Search className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-slate-500" />
   462→          <Input
   463→            value={searchInput}
   464→            onChange={(e) => setSearchInput(e.target.value)}
   465→            placeholder="Search ideas by title…"
   466→            className="h-9 border-slate-700/60 bg-slate-900/60 pl-9 pr-8 text-sm text-slate-100 placeholder:text-slate-500 focus-visible:border-violet-500/50 focus-visible:ring-violet-500/20"
   467→          />
   468→          {searchInput && (
   469→            <button
   470→              type="button"
   471→              onClick={() => setSearchInput('')}
   472→              className="absolute right-2 top-1/2 -translate-y-1/2 rounded p-1 text-slate-500 hover:bg-slate-800 hover:text-slate-300"
   473→              aria-label="Clear search"
   474→            >
   475→              <X className="size-3.5" />
   476→            </button>
   477→          )}
   478→        </div>
   479→      </div>
   480→
   481→      {/* ─── Filter Row ─── */}
   482→      <div className="mb-3 flex flex-wrap items-center gap-2">
   483→        {/* Type pills */}
   484→        <div className="flex items-center rounded-lg border border-slate-700/60 bg-slate-900/40 p-0.5">
   485→          {(['all', 'short', 'longform'] as const).map((t) => {
   486→            const isActive = typeFilter === t
   487→            const label =
   488→              t === 'all' ? 'All' : t === 'short' ? 'Short' : 'Long'
   489→            return (
   490→              <button
   491→                key={t}
   492→                type="button"
   493→                onClick={() => setTypeFilter(t)}
   494→                className={cn(
   495→                  'relative rounded-md px-3 py-1 text-xs font-medium transition-colors',
   496→                  isActive
   497→                    ? 'text-white'
   498→                    : 'text-slate-400 hover:text-slate-200',
   499→                )}
   500→              >
   501→                {isActive && (
   502→                  <motion.span
   503→                    layoutId="type-pill-bg"
   504→                    className="absolute inset-0 rounded-md bg-gradient-to-r from-violet-500/80 to-cyan-500/80"
   505→                    transition={{ type: 'spring', duration: 0.3, bounce: 0.2 }}
   506→                  />
   507→                )}
   508→                <span className="relative z-10">{label}</span>
   509→              </button>
   510→            )
   511→          })}
   512→        </div>
   513→
   514→        {/* Status filter */}
   515→        <Select
   516→          value={statusFilter}
   517→          onValueChange={(v) => setStatusFilter(v as 'all' | IdeaStatus)}
   518→        >
   519→          <SelectTrigger
   520→            size="sm"
   521→            className="h-8 w-[150px] border-slate-700/60 bg-slate-900/60 text-xs text-slate-200 hover:bg-slate-800/60"
   522→          >
   523→            <Filter className="size-3.5 text-slate-500" />
   524→            <SelectValue placeholder="Status" />
   525→          </SelectTrigger>
   526→          <SelectContent className="border-slate-700/60 bg-slate-900 text-slate-200">
   527→            <SelectItem value="all">All statuses</SelectItem>
   528→            <Separator className="bg-slate-700/60" />
   529→            {(Object.keys(STATUS_CONFIG) as IdeaStatus[]).map((s) => (
   530→              <SelectItem key={s} value={s}>
   531→                <span className="flex items-center gap-2">
   532→                  <span
   533→                    className={cn(

   535→                      STATUS_CONFIG[s].dotClass,
   536→                    )}
   537→                  />
   538→                  {STATUS_CONFIG[s].label}
   539→                </span>

   543→        </Select>

   545→        {/* Pillar filter */}
   546→        <Select

   549→        >
   550→          <SelectTrigger
