   138→function stateColor(state: string) {
   139→  const activeStates = ['running', 'researching_niches', 'creating_strategy', 'researching_topic', 'writing_script', 'producing_video', 'reviewing', 'uploading', 'cycle_complete']
   140→  if (activeStates.includes(state)) return { dot: 'bg-emerald-500', ring: 'ring-emerald-500/30', shadow: 'shadow-emerald-500/20', text: 'text-emerald-400', label: 'Running' }
   141→  if (state === 'error') return { dot: 'bg-red-500', ring: 'ring-red-500/30', shadow: 'shadow-red-500/20', text: 'text-red-400', label: 'Error' }
   142→  if (state === 'paused') return { dot: 'bg-amber-500', ring: 'ring-amber-500/30', shadow: 'shadow-amber-500/20', text: 'text-amber-400', label: 'Paused' }
   143→  return { dot: 'bg-slate-400', ring: 'ring-slate-400/30', shadow: 'shadow-slate-400/20', text: 'text-slate-400', label: 'Idle' }
   144→}

   146→function actionColor(action: string) {
   147→  // New categorized actions
   148→  if (action === 'emergency_stop') return 'text-red-400 bg-red-500/10 border-red-500/30'
   149→  if (action === 'mode_change') return 'text-amber-400 bg-amber-500/10 border-amber-500/30'
   150→  if (action === 'upload') return 'text-cyan-400 bg-cyan-500/10 border-cyan-500/30'
   151→  if (action === 'strategy_change') return 'text-violet-400 bg-violet-500/10 border-violet-500/30'
   152→  // Legacy / fallback keyword matching
   153→  if (action.includes('error') || action.includes('fail')) return 'text-red-400 bg-red-500/10 border-red-500/30'
   154→  if (action.includes('upload') || action.includes('complete')) return 'text-emerald-400 bg-emerald-500/10 border-emerald-500/30'
   155→  if (action.includes('produce') || action.includes('render')) return 'text-cyan-400 bg-cyan-500/10 border-cyan-500/30'
   156→  if (action.includes('script') || action.includes('write')) return 'text-amber-400 bg-amber-500/10 border-amber-500/30'
   157→  if (action.includes('research') || action.includes('niche')) return 'text-cyan-400 bg-cyan-500/10 border-cyan-500/30'
   158→  if (action.includes('review') || action.includes('quality')) return 'text-rose-400 bg-rose-500/10 border-rose-500/30'
   159→  return 'text-slate-400 bg-slate-500/10 border-slate-500/30'
   160→}

   336→// ─── Main Dashboard ─────────────────────────────────────────────────
   337→export default function Dashboard() {
   338→  const [status, setStatus] = useState<AgentStatus | null>(null)
   339→  const [pipeline, setPipeline] = useState<PipelineData | null>(null)
   340→  const [channel, setChannel] = useState<ChannelData | null>(null)
   341→  const [jobs, setJobs] = useState<any[]>([])
   342→  const [logs, setLogs] = useState<any[]>([])
   343→  const [analytics, setAnalytics] = useState<AnalyticsData | null>(null)
   344→  const [loading, setLoading] = useState(false)
   345→  const [activeTab, setActiveTab] = useState('overview')
   346→  const [lastPoll, setLastPoll] = useState<Date>(new Date())
   347→  const [previewVideoId, setPreviewVideoId] = useState<string | null>(null)
   348→  const [initialLoaded, setInitialLoaded] = useState(false)
   349→
   350→  // ── Polling ─────────────────────────────────────────────────────
   351→  const fetchStatus = useCallback(async () => {
   352→    try {
   353→      const res = await fetch('/api/agent/status')
   354→      if (res.ok) setStatus(await res.json())
   355→    } catch {}
   356→  }, [])
   357→
   358→  const fetchPipeline = useCallback(async () => {
   359→    try {
   360→      const res = await fetch('/api/data/pipeline')
   361→      if (res.ok) setPipeline(await res.json())
   362→    } catch {}
   363→  }, [])
   364→
   365→  const fetchChannel = useCallback(async () => {
   366→    try {
   367→      const res = await fetch('/api/data/channel')
   368→      if (res.ok) setChannel(await res.json())
   369→    } catch {}
   370→  }, [])
   371→
   372→  const fetchJobs = useCallback(async () => {
   373→    try {
   374→      const res = await fetch('/api/data/jobs')
   375→      if (res.ok) setJobs(await res.json())
   376→    } catch {}
   377→  }, [])
   378→
   379→  const fetchLogs = useCallback(async () => {
   380→    try {
   381→      const res = await fetch('/api/data/audit-logs')
   382→      if (res.ok) setLogs(await res.json())
   383→    } catch {}
   384→  }, [])
   385→
   386→  const fetchAnalytics = useCallback(async () => {
   387→    try {
   388→      const res = await fetch('/api/data/analytics')
   389→      if (res.ok) setAnalytics(await res.json())
   390→    } catch {}
   391→  }, [])
   392→
   393→  const pollAll = useCallback(async () => {
   394→    await Promise.all([fetchStatus(), fetchPipeline(), fetchChannel(), fetchJobs(), fetchLogs(), fetchAnalytics()])
   395→    setLastPoll(new Date())
   396→    setInitialLoaded(true)
   397→  }, [fetchStatus, fetchPipeline, fetchChannel, fetchJobs, fetchLogs, fetchAnalytics])
   398→
   399→  useEffect(() => {
   400→    pollAll()
   401→    const interval = setInterval(pollAll, 5000)
   402→    return () => clearInterval(interval)
   403→  }, [pollAll])
   404→
   405→  // ── Commands ────────────────────────────────────────────────────
   406→  const sendCommand = async (command: string, extra?: any) => {
   407→    setLoading(true)
   408→    try {
   409→      const res = await fetch('/api/agent/command', {
   410→        method: 'POST',
   411→        headers: { 'Content-Type': 'application/json' },
   412→        body: JSON.stringify({ command, ...extra }),
   413→      })
   414→      const data = await res.json()
   415→      await fetchStatus()
   416→      return data
   417→    } catch (e: any) {
   418→      console.error(e)
   419→    } finally {
   420→      setLoading(false)
   421→    }
   422→  }
   423→
   424→  // ── Computed ────────────────────────────────────────────────────
   425→  const totalPipeline = status ? Object.values(status.pipeline).reduce((a: number, b: number) => a + b, 0) : 0
   426→  const pipelineChartData = useMemo(() => {
   427→    if (!logs.length) return []
   428→    // Generate synthetic time-series from pipeline counts for the area chart
   429→    const now = Date.now()
   430→    return Array.from({ length: 12 }, (_, i) => ({
   431→      time: new Date(now - (11 - i) * 3600000).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
   432→      ideas: Math.max(0, (status?.pipeline.ideas || 0) - Math.floor(Math.random() * i * 0.3)),
   433→      produced: Math.max(0, (status?.pipeline.producing || 0) + Math.floor(Math.random() * 2)),
   434→      uploaded: Math.max(0, (status?.pipeline.uploaded || 0) - Math.floor(Math.random() * i * 0.1)),
   435→    }))
   436→  }, [status?.pipeline, logs.length])
   437→
   438→  const nicheBarData = useMemo(() => {
   439→    if (!channel?.niches?.length) return []
   440→    return channel.niches.slice(0, 10).map((n: any) => ({
   441→      name: n.nicheName.length > 18 ? n.nicheName.slice(0, 18) + '...' : n.nicheName,
   442→      score: n.compositeScore || 0,
   443→      selected: n.isSelected,
   444→    }))
   445→  }, [channel?.niches])
   446→
   447→  // ── Render ──────────────────────────────────────────────────────
   448→  return (
   449→    <div className="min-h-screen bg-slate-950 text-slate-100 flex flex-col">
