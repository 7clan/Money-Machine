   218→   218→// ─── Animated Counter Hook ───────────────────────────────────────────
   219→   219→function useAnimatedCounter(end: number, duration: number = 800) {
   220→   220→  const [count, setCount] = useState(end)
   221→   221→  const prevEnd = useRef(end)
   222→   222→  useEffect(() => {
   223→   223→    if (prevEnd.current === end) return
   224→   224→    const start = prevEnd.current
   225→   225→    const startTime = performance.now()
   226→   226→    const step = (now: number) => {
   227→   227→      const progress = Math.min((now - startTime) / duration, 1)
   228→   228→      const eased = 1 - Math.pow(1 - progress, 3) // easeOutCubic
   229→   229→      setCount(Math.round(start + (end - start) * eased))
   230→   230→      if (progress < 1) requestAnimationFrame(step)
   231→   231→    }
   232→   232→    requestAnimationFrame(step)
   233→   233→    prevEnd.current = end
   234→   234→    return () => {}
   235→   235→  }, [end, duration])
   236→   236→  return count
   237→   237→}

   292→   292→function PipelineFlow({ pipeline }: { pipeline: AgentStatus['pipeline'] | null }) {
   293→   293→  if (!pipeline) return null
   294→   294→  const total = pipeline ? Object.values(pipeline).reduce((a: number, b: number) => a + b, 0) : 0
   295→   295→  return (
   296→   296→    <div className="flex items-center gap-0 overflow-x-auto py-2 px-1">
   297→   297→      {PIPELINE_STAGES.map((stage, i) => {
   298→   298→        const count = pipeline[stage.key as keyof typeof pipeline] || 0
   299→   299→        const pct = total > 0 ? Math.round((count / total) * 100) : 0
   300→   300→        const Icon = stage.icon
   301→   301→        return (
   302→   302→          <React.Fragment key={stage.key}>
   303→   303→            <motion.div
   304→   304→              initial={{ opacity: 0, scale: 0.8 }}
   305→   305→              animate={{ opacity: 1, scale: 1 }}
   306→   306→              transition={{ delay: i * 0.08, duration: 0.3 }}
   307→   307→              className={`flex flex-col items-center min-w-[80px] p-3 rounded-xl border ${stage.border} ${stage.bg} transition-all duration-300 hover:scale-105 hover:shadow-lg`}
   308→   308→            >
   309→   309→              <Icon className={`w-5 h-5 ${stage.textColor} mb-1`} />
   310→   310→              <span className={`text-xl font-bold ${stage.textColor}`}>{count}</span>
   311→   311→              <span className="text-[10px] text-slate-400 mt-0.5">{stage.label}</span>
   312→   312→              {pct > 0 && (
   313→   313→                <span className="text-[9px] text-slate-500 font-mono mt-0.5">{pct}%</span>
   314→   314→              )}
   315→   315→            </motion.div>
   316→   316→            {i < PIPELINE_STAGES.length - 1 && (
   317→   317→              <motion.div
   318→   318→                initial={{ opacity: 0, x: -8 }}
   319→   319→                animate={{ opacity: 1, x: 0 }}
   320→   320→                transition={{ delay: i * 0.08 + 0.04, duration: 0.3 }}
   321→   321→                className="flex items-center px-1"
   322→   322→              >
   323→   323→                <ChevronRight className="w-4 h-4 text-slate-600" />
   324→   324→              </motion.div>
   325→   325→            )}
   326→   326→          </React.Fragment>
   327→   327→        )
   328→   328→      })}
   329→   329→    </div>
   330→   330→  )
