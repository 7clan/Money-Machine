   174→function formatDuration(seconds?: number | null): string {
   175→  if (seconds == null || Number.isNaN(seconds)) return '—'
   176→  const s = Math.round(seconds)
   177→  const m = Math.floor(s / 60)
   178→  const r = s % 60
   179→  return `${m}:${r.toString().padStart(2, '0')}`
   180→}
   181→
   182→function formatFileSize(bytes?: number | null): string {
   183→  if (!bytes) return '—'
   184→  if (bytes < 1024) return `${bytes} B`
   185→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   186→  if (bytes < 1024 * 1024 * 1024) return `${(bytes / (1024 * 1024)).toFixed(1)} MB`
   187→  return `${(bytes / (1024 * 1024 * 1024)).toFixed(2)} GB`
   188→}

   190→// Map pillar.color (any CSS color) into a safe tailwind text/bg class set.
   191→// Falls back to the violet→cyan brand gradient.
   192→function pillarStyle(color?: string | null): { text: string; bg: string; border: string } {
   193→  const c = (color || '').toLowerCase()
   194→  switch (c) {
   195→    case 'violet':
   196→    case '#8b5cf6':
   197→    case 'purple':
   198→      return { text: 'text-violet-300', bg: 'bg-violet-500/10', border: 'border-violet-500/30' }
   199→    case 'cyan':
   200→    case '#06b6d4':
   201→      return { text: 'text-cyan-300', bg: 'bg-cyan-500/10', border: 'border-cyan-500/30' }
   202→    case 'emerald':
   203→    case '#10b981':
   204→    case 'green':
   205→      return { text: 'text-emerald-300', bg: 'bg-emerald-500/10', border: 'border-emerald-500/30' }
   206→    case 'amber':
   207→    case '#f59e0b':
   208→    case 'yellow':
   209→      return { text: 'text-amber-300', bg: 'bg-amber-500/10', border: 'border-amber-500/30' }
   210→    case 'rose':
   211→    case '#f43f5e':
   212→    case 'red':
   213→      return { text: 'text-rose-300', bg: 'bg-rose-500/10', border: 'border-rose-500/30' }
   214→    default:
   215→      return { text: 'text-violet-300', bg: 'bg-violet-500/10', border: 'border-violet-500/30' }
   216→  }
   217→}

   219→// Color-code VideoProject.status.
   220→function statusStyle(status: string): { label: string; text: string; bg: string; border: string; dot: string } {
   221→  const s = (status || '').toLowerCase()
   222→  if (['approved', 'uploaded', 'published', 'completed'].includes(s)) {
   223→    return { label: status, text: 'text-emerald-300', bg: 'bg-emerald-500/10', border: 'border-emerald-500/30', dot: 'bg-emerald-400' }
   224→  }
   225→  if (['failed', 'rejected', 'error'].includes(s)) {
   226→    return { label: status, text: 'text-rose-300', bg: 'bg-rose-500/10', border: 'border-rose-500/30', dot: 'bg-rose-400' }
   227→  }
   228→  if (['rendering', 'editing', 'recording', 'collecting_assets', 'storyboarding', 'planning', 'uploading'].includes(s)) {
   229→    return { label: status, text: 'text-cyan-300', bg: 'bg-cyan-500/10', border: 'border-cyan-500/30', dot: 'bg-cyan-400' }
   230→  }
   231→  if (['review', 'reviewing'].includes(s)) {
   232→    return { label: status, text: 'text-amber-300', bg: 'bg-amber-500/10', border: 'border-amber-500/30', dot: 'bg-amber-400' }
   233→  }
   234→  return { label: status, text: 'text-slate-300', bg: 'bg-slate-500/10', border: 'border-slate-500/30', dot: 'bg-slate-400' }
   235→}

   237→// Parse a JSON-string array (issues, etc.) safely.
   238→function parseJsonArray(raw?: string | null): string[] {
   239→  if (!raw) return []
   240→  try {
   241→    const v = JSON.parse(raw)
   242→    if (Array.isArray(v)) return v.map((x) => String(x))
   243→    return [String(v)]
   244→  } catch {
   245→    return raw.split('\n').map((s) => s.trim()).filter(Boolean)
   246→  }
   247→}
