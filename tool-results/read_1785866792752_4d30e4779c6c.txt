   182→  if (state === 'error') return { dot: 'bg-red-500', ring: 'ring-red-500/30', shadow: 'shadow-red-500/20', text: 'text-red-400', label: 'Error' }
   183→  if (state === 'paused') return { dot: 'bg-amber-500', ring: 'ring-amber-500/30', shadow: 'shadow-amber-500/20', text: 'text-amber-400', label: 'Paused' }
   184→  return { dot: 'bg-slate-400', ring: 'ring-slate-400/30', shadow: 'shadow-slate-400/20', text: 'text-slate-400', label: 'Idle' }
   185→}

   230→// ─── Animated Counter Hook ───────────────────────────────────────────
   231→function useAnimatedCounter(end: number, duration: number = 800) {
   232→  const [count, setCount] = useState(end)
   233→  const prevEnd = useRef(end)
   234→  useEffect(() => {
   235→    if (prevEnd.current === end) return
   236→    const start = prevEnd.current
   237→    const startTime = performance.now()
   238→    const step = (now: number) => {
   239→      const progress = Math.min((now - startTime) / duration, 1)
   240→      const eased = 1 - Math.pow(1 - progress, 3) // easeOutCubic
   241→      setCount(Math.round(start + (end - start) * eased))
   242→      if (progress < 1) requestAnimationFrame(step)
   243→    }
   244→    requestAnimationFrame(step)
   245→    prevEnd.current = end
   246→    return () => {}
   247→  }, [end, duration])
   248→  return count
   249→}

   312→function PipelineFlow({ pipeline }: { pipeline: AgentStatus['pipeline'] | null }) {
   313→  if (!pipeline) return null
   314→  const total = pipeline ? Object.values(pipeline).reduce((a: number, b: number) => a + b, 0) : 0
   315→  return (
   316→    <div className="flex items-center gap-0 overflow-x-auto py-2 px-1">
   317→      {PIPELINE_STAGES.map((stage, i) => {
   318→        const count = pipeline[stage.key as keyof typeof pipeline] || 0
   319→        const pct = total > 0 ? Math.round((count / total) * 100) : 0
   320→        const Icon = stage.icon
   321→        const isEmpty = count === 0
   322→        return (
   323→          <React.Fragment key={stage.key}>
   324→            <motion.div
   325→              initial={{ opacity: 0, scale: 0.8 }}
   326→              animate={{ opacity: 1, scale: 1 }}
   327→              transition={{ delay: i * 0.08, duration: 0.3 }}
   328→              className={`flex flex-col items-center min-w-[80px] p-3 rounded-xl border ${stage.border} ${stage.bg} transition-all duration-300 hover:scale-105 hover:shadow-lg ${isEmpty ? 'opacity-60' : ''}`}
   329→            >
   330→              <Icon className={`w-5 h-5 ${stage.textColor} mb-1`} />
   331→              <span className={`text-xl font-bold ${stage.textColor} font-tabular-nums`}>{isEmpty ? '+' : count}</span>
   332→              <span className="text-[10px] text-slate-400 mt-0.5">{stage.label}</span>
   333→              {pct > 0 && (
   334→                <span className="text-[10px] text-slate-400 font-mono mt-0.5">{pct}%</span>
   335→              )}
   336→            </motion.div>
   337→            {i < PIPELINE_STAGES.length - 1 && (
   338→              <motion.div
   339→                initial={{ opacity: 0, x: -8 }}
   340→                animate={{ opacity: 1, x: 0 }}
   341→                transition={{ delay: i * 0.08 + 0.04, duration: 0.3 }}
   342→                className="flex items-center px-1"
   343→              >
   344→                <ChevronRight className="w-4 h-4 text-slate-500 group-hover/stage:text-slate-400 transition-colors" />
   345→              </motion.div>
   346→            )}
   347→          </React.Fragment>
   348→        )
   349→      })}
   350→    </div>
   351→  )
   352→}
