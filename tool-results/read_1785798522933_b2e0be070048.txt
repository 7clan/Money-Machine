   167→function stateColor(state: string) {
   168→  const activeStates = ['running', 'researching_niches', 'creating_strategy', 'researching_topic', 'writing_script', 'producing_video', 'reviewing', 'uploading', 'cycle_complete']
   169→  if (activeStates.includes(state)) return { dot: 'bg-emerald-500', ring: 'ring-emerald-500/30', shadow: 'shadow-emerald-500/20', text: 'text-emerald-400', label: 'Running' }
   170→  if (state === 'error') return { dot: 'bg-red-500', ring: 'ring-red-500/30', shadow: 'shadow-red-500/20', text: 'text-red-400', label: 'Error' }
   171→  if (state === 'paused') return { dot: 'bg-amber-500', ring: 'ring-amber-500/30', shadow: 'shadow-amber-500/20', text: 'text-amber-400', label: 'Paused' }
   172→  return { dot: 'bg-slate-400', ring: 'ring-slate-400/30', shadow: 'shadow-slate-400/20', text: 'text-slate-400', label: 'Idle' }
   173→}

   175→function actionColor(action: string) {
   176→  // New categorized actions
   177→  if (action === 'emergency_stop') return 'text-red-400 bg-red-500/10 border-red-500/30'
   178→  if (action === 'mode_change') return 'text-amber-400 bg-amber-500/10 border-amber-500/30'
   179→  if (action === 'upload') return 'text-cyan-400 bg-cyan-500/10 border-cyan-500/30'
   180→  if (action === 'strategy_change') return 'text-violet-400 bg-violet-500/10 border-violet-500/30'
   181→  // Legacy / fallback keyword matching
   182→  if (action.includes('error') || action.includes('fail')) return 'text-red-400 bg-red-500/10 border-red-500/30'
   183→  if (action.includes('upload') || action.includes('complete')) return 'text-emerald-400 bg-emerald-500/10 border-emerald-500/30'
   184→  if (action.includes('produce') || action.includes('render')) return 'text-cyan-400 bg-cyan-500/10 border-cyan-500/30'
   185→  if (action.includes('script') || action.includes('write')) return 'text-amber-400 bg-amber-500/10 border-amber-500/30'
   186→  if (action.includes('research') || action.includes('niche')) return 'text-cyan-400 bg-cyan-500/10 border-cyan-500/30'
   187→  if (action.includes('review') || action.includes('quality')) return 'text-rose-400 bg-rose-500/10 border-rose-500/30'
   188→  return 'text-slate-400 bg-slate-500/10 border-slate-500/30'
   189→}

   232→function StatusCard({ icon: Icon, label, value, sub, trend, color = 'text-emerald-400', valueSuffix, hint }: {
   233→  icon: any; label: string; value: string | number; sub?: string; trend?: 'up' | 'down' | 'flat'; color?: string; valueSuffix?: string; hint?: string
   234→}) {
   235→  const bgForColor = (c: string) => c === 'text-emerald-400' ? 'bg-emerald-500/10' : c === 'text-blue-400' ? 'bg-blue-500/10' : c === 'text-amber-400' ? 'bg-amber-500/10' : c === 'text-violet-400' ? 'bg-violet-500/10' : c === 'text-cyan-400' ? 'bg-cyan-500/10' : c === 'text-rose-400' ? 'bg-rose-500/10' : 'bg-slate-500/10'
   236→  return (
   237→    <GradientCard>
   238→      <div className="p-4 group cursor-default" title={hint}>
   239→        <div className="flex items-start justify-between">
   240→          <div className={`p-2 rounded-lg ${bgForColor(color)} transition-transform duration-300 group-hover:scale-110`}>
   241→            <Icon className={`w-4 h-4 ${color}`} />
   242→          </div>
   243→          {trend && (
   244→            <div className={`flex items-center text-xs ${trend === 'up' ? 'text-emerald-400' : trend === 'down' ? 'text-red-400' : 'text-slate-400'}`}>
   245→              {trend === 'up' ? <ArrowUpRight className="w-3 h-3" /> : trend === 'down' ? <ArrowDownRight className="w-3 h-3" /> : <Minus className="w-3 h-3" />}
   246→            </div>
   247→          )}
   248→        </div>
   249→        <div className="mt-3">
   250→          <div className="flex items-baseline gap-1">
   251→            <p className="text-xl sm:text-2xl font-bold tracking-tight truncate" title={String(value)}>{value}</p>
   252→            {valueSuffix && <span className={`text-sm font-semibold ${color}`}>{valueSuffix}</span>}
   253→          </div>
   254→          <p className="text-xs text-slate-400 mt-0.5">{label}</p>

   256→        </div>
   257→      </div>
   258→    </GradientCard>
