   144→  if (raw === 'uploading') return 'uploaded'
   145→  return 'producing'

   185→function toTimestamp(value?: string | Date | null): number {
   186→  if (!value) return 0
   187→  const d = value instanceof Date ? value : new Date(value)
   188→  const t = d.getTime()
   189→  return Number.isNaN(t) ? 0 : t
   190→}
   191→
   192→function relativeTime(value?: string | Date | null): string {
   193→  const t = toTimestamp(value)
   194→  if (!t) return '—'
   195→  const diff = Date.now() - t
   196→  if (diff < 0) return 'just now'
   197→  const sec = Math.floor(diff / 1000)
   198→  if (sec < 60) return `${sec}s ago`
   199→  const min = Math.floor(sec / 60)
   200→  if (min < 60) return `${min}m ago`
   201→  const hr = Math.floor(min / 60)
   202→  if (hr < 24) return `${hr}h ago`
   203→  const day = Math.floor(hr / 24)
   204→  if (day < 7) return `${day}d ago`
   205→  const week = Math.floor(day / 7)
   206→  if (week < 4) return `${week}w ago`
   207→  const month = Math.floor(day / 30)
   208→  if (month < 12) return `${month}mo ago`
   209→  const year = Math.floor(day / 365)
   210→  return `${year}y ago`
   211→}
   212→
   213→function formatDuration(seconds?: number | null): string {
   214→  if (seconds === null || seconds === undefined || Number.isNaN(seconds)) {
   215→    return '—'
   216→  }
   217→  if (seconds < 60) return `${seconds.toFixed(0)}s`
   218→  const m = Math.floor(seconds / 60)
   219→  const s = Math.floor(seconds % 60)
   220→  return `${m}:${s.toString().padStart(2, '0')}`
   221→}
   222→
   223→function formatFileSize(bytes?: number | null): string {
   224→  if (!bytes || bytes <= 0) return '—'
   225→  if (bytes < 1024) return `${bytes} B`
   226→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`
   227→  if (bytes < 1024 * 1024 * 1024) return `${(bytes / 1024 / 1024).toFixed(1)} MB`
   228→  return `${(bytes / 1024 / 1024 / 1024).toFixed(2)} GB`
   229→}
   230→
   231→function truncate(text: string, max: number): string {
   232→  if (text.length <= max) return text
   233→  return text.slice(0, max - 1).trimEnd() + '…'
   234→}

   249→  )
   250→  const [sortKey, setSortKey] = useState<SortKey>('updated_desc')
   251→
   252→  // ─── Per-card re-render flow (single project) ─────────────────────
   253→  // The bulk "re-render" action lives below; this state powers the
   254→  // per-card button that calls /api/agent/rerender for one project.
   255→  const [rerenderConfirmId, setRerenderConfirmId] = useState<string | null>(null)
   256→  const [rerenderingIds, setRerenderingIds] = useState<Set<string>>(new Set())
   257→
   258→  // ─── Bulk selection state ──────────────────────────────────────────
   259→  const [selectMode, setSelectMode] = useState(false)
   260→  const [selectedIds, setSelectedIds] = useState<Set<string>>(new Set())
   261→  const [bulkLoading, setBulkLoading] = useState<BulkActionType | null>(null)
   262→  const [confirmDialog, setConfirmDialog] = useState<
   263→    null | 'delete' | 're-render'
   264→  >(null)
   265→
   266→  // Debounced search (200ms)
   267→  useEffect(() => {
   268→    const handle = setTimeout(() => {
   269→      setDebouncedSearch(searchInput.trim().toLowerCase())
   270→    }, 200)
   271→    return () => clearTimeout(handle)
   272→  }, [searchInput])
   273→
   274→  const filteredProjects = useMemo(() => {
   275→    const out: VideoProject[] = []
   276→    for (const project of projects) {
   277→      const title = (project.title ?? '').toLowerCase()
   278→      if (debouncedSearch && !title.includes(debouncedSearch)) continue
   279→      if (statusFilter !== 'all') {
   280→        const s = coerceStatus(project.status)
   281→        if (s !== statusFilter) continue
   282→      }
   283→      out.push(project)
   284→    }
   285→
   286→    out.sort((a, b) => {
   287→      switch (sortKey) {
   288→        case 'updated_asc':
   289→          return toTimestamp(a.updatedAt) - toTimestamp(b.updatedAt)
   290→        case 'updated_desc':
   291→          return toTimestamp(b.updatedAt) - toTimestamp(a.updatedAt)
   292→        case 'duration_asc': {
   293→          const da = a.duration ?? 0
   294→          const db = b.duration ?? 0
   295→          return da - db
   296→        }
   297→        case 'duration_desc': {
   298→          const da = a.duration ?? 0
   299→          const db = b.duration ?? 0
   300→          return db - da
   301→        }
   302→        case 'status': {
   303→          const sa = coerceStatus(a.status)
   304→          const sb = coerceStatus(b.status)
   305→          return sa.localeCompare(sb)
   306→        }
   307→        default:
   308→          return 0
   309→      }
   310→    })
   311→    return out
   312→  }, [projects, debouncedSearch, statusFilter, sortKey])
   313→
   314→  const activeFilterCount = useMemo(() => {
   315→    let n = 0
   316→    if (debouncedSearch) n++
   317→    if (statusFilter !== 'all') n++
   318→    return n
   319→  }, [debouncedSearch, statusFilter])
   320→
   321→  // Prune stale IDs from selection when the projects list changes
   322→  // (e.g. after a bulk delete or after the parent refetches).
   323→  useEffect(() => {
   324→    if (selectedIds.size === 0) return
   325→    const validIds = new Set(projects.map((p) => p.id))
   326→    let changed = false
   327→    const next = new Set<string>()
   328→    for (const id of selectedIds) {
   329→      if (validIds.has(id)) {
   330→        next.add(id)
   331→      } else {
   332→        changed = true
   333→      }
   334→    }
   335→    if (changed) {
   336→      setSelectedIds(next)
   337→      if (next.size === 0) setSelectMode(false)
   338→    }
   339→  }, [projects, selectedIds])
   340→
   341→  const clearFilters = useCallback(() => {
   342→    setSearchInput('')
   343→    setDebouncedSearch('')
   344→    setStatusFilter('all')
   345→    setSortKey('updated_desc')
   346→  }, [])
   347→
   348→  // ─── Bulk action handlers ──────────────────────────────────────────
   349→  const toggleSelectMode = useCallback(() => {
   350→    setSelectMode((prev) => {
   351→      const next = !prev
   352→      if (!next) setSelectedIds(new Set())
   353→      return next
   354→    })
   355→  }, [])
   356→
   357→  const toggleSelect = useCallback((id: string) => {
   358→    setSelectedIds((prev) => {
   359→      const next = new Set(prev)
   360→      if (next.has(id)) next.delete(id)
   361→      else next.add(id)
   362→      return next
   363→    })
   364→  }, [])
   365→
   366→  const selectAllFiltered = useCallback(() => {
   367→    setSelectedIds(new Set(filteredProjects.map((p) => p.id)))
   368→  }, [filteredProjects])
   369→
   370→  const clearSelection = useCallback(() => {
   371→    setSelectedIds(new Set())
   372→  }, [])
   373→
   374→  const executeBulkAction = useCallback(
   375→    async (
   376→      action: BulkActionType,
   377→      payload?: Record<string, unknown>,
   378→    ) => {
   379→      if (selectedIds.size === 0) return
   380→      setBulkLoading(action)
   381→      const ids = Array.from(selectedIds)
   382→      try {
   383→        const res = await fetch('/api/data/projects/bulk', {
   384→          method: 'POST',
   385→          headers: { 'Content-Type': 'application/json' },
   386→          body: JSON.stringify({ action, projectIds: ids, payload }),
   387→        })
   388→        const data = (await res.json().catch(() => ({}))) as {
   389→          ok?: boolean
   390→          affected?: number
   391→          error?: string
   392→        }
   393→        if (!res.ok || !data.ok) {
   394→          throw new Error(
   395→            data.error || `Bulk ${action} failed (HTTP ${res.status})`,
   396→          )
   397→        }
   398→        const affected = data.affected ?? 0
   399→        toast({
   400→          type: 'success',
   401→          title: `Bulk ${action} complete`,
   402→          description: `${affected} project${affected === 1 ? '' : 's'} affected.`,
   403→        })
   404→        // Clear local selection first, then notify parent so it can refresh.
   405→        setSelectedIds(new Set())
   406→        setSelectMode(false)
   407→        setConfirmDialog(null)
   408→        onBulkAction?.()
   409→      } catch (err) {
   410→        toast({
   411→          type: 'error',
   412→          title: `Bulk ${action} failed`,
   413→          description: err instanceof Error ? err.message : 'Unknown error',
   414→          duration: 5000,
   415→        })
   416→      } finally {
   417→        setBulkLoading(null)
   418→      }
   419→    },
   420→    [selectedIds, toast, onBulkAction],

   422→

   456→          message?: string
   457→          error?: string
   458→        }

   461→            data?.message || data?.error || `Re-render failed (${res.status})`

   463→            type: 'error',
   464→            title: 'Re-render failed',
   465→            description: errMsg,
   466→            duration: 5000,

   469→            const next = new Set(prev)

   476→          type: 'success',
   477→          title: 'Re-render started',
   478→          description: data?.message || 'A new script version is being generated.',

   483→      } catch (err) {

   485→          type: 'error',
   486→          title: 'Network error',
   487→          description: err instanceof Error ? err.message : 'Failed to reach server',
   488→          duration: 5000,
   489→        })
   490→        setRerenderingIds((prev) => {
   491→          const next = new Set(prev)
   492→          next.delete(projectId)
   493→          return next
   494→        })
   495→      }
   496→    },
   497→    [toast, updateToast, onBulkAction],

   500→  const isLoading = projects.length === 0
   501→
   502→  // ─── Render ────────────────────────────────────────────────────────
   503→
   504→  return (
   505→    <div
   506→      className={cn(
   507→        'w-full rounded-xl bg-slate-950 p-3 sm:p-4',
   508→        'border border-slate-800/50',
   509→        className,
   510→      )}
   511→    >
   512→      {/* ─── Header / Search ─── */}
   513→      <div className="mb-3 flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   514→        <div className="flex items-center gap-2">
   515→          <div className="flex size-8 items-center justify-center rounded-lg bg-gradient-to-br from-emerald-500/20 to-cyan-500/20 border border-emerald-500/30">
   516→            <Film className="size-4 text-emerald-300" />
   517→          </div>
   518→          <div>
   519→            <h3 className="text-sm font-semibold text-slate-100">
   520→              Video Project Explorer
   521→            </h3>
   522→            <p className="text-[11px] text-slate-400">
   523→              Produce, approve &amp; manage rendered videos
   524→            </p>
   525→          </div>
   526→        </div>
   527→
   528→        <div className="flex items-center gap-2">
   529→          <div className="relative w-full sm:w-72">
   530→            <Search className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-slate-500" />
   531→            <Input
   532→              value={searchInput}
   533→              onChange={(e) => setSearchInput(e.target.value)}
   534→              placeholder="Search projects by title…"
   535→              className="h-9 border-slate-700/60 bg-slate-900/60 pl-9 pr-8 text-sm text-slate-100 placeholder:text-slate-500 focus-visible:border-emerald-500/50 focus-visible:ring-emerald-500/20"
   536→            />
   537→            {searchInput && (
   538→              <button
   539→                type="button"
   540→                onClick={() => setSearchInput('')}
   541→                className="absolute right-2 top-1/2 -translate-y-1/2 rounded p-1 text-slate-500 hover:bg-slate-800 hover:text-slate-300"
   542→                aria-label="Clear search"

   544→                <X className="size-3.5" />

   547→          </div>
   548→
   549→          {/* Select toggle */}
   550→          <Tooltip>

   559→                  selectMode
   560→                    ? 'border-transparent bg-gradient-to-r from-emerald-500 to-cyan-500 text-white hover:from-emerald-600 hover:to-cyan-600'
   561→                    : 'border-slate-700/60 bg-slate-900/60 text-slate-300 hover:bg-slate-800/60 hover:text-slate-100',

   571→            </TooltipTrigger>
   572→            <TooltipContent className="bg-slate-800 text-slate-100">

   575→          </Tooltip>

   579→      {/* ─── Filter Row: status pills + sort ─── */}
